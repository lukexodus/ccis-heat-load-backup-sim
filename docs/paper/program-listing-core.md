```python
    Static parameters and boundary limits. Dispatch logic is handled in Phase 2 (BAT-001/002).

    distinct from and additional to the non-discretionary 10% min SoC protection floor from [D-09].

    capacity_kwh: float
    round_trip_efficiency: float = 0.92      # 92% round-trip efficiency per [D-09]
    usable_dod: float = 0.90                 # 90% usable depth of discharge per [D-09]
    min_soc_pct: float = 0.10                # 10% minimum SoC reserve per [D-09]
    max_charge_c_rate: float = 0.5           # 0.5C maximum charging rate per [D-09]
    max_discharge_c_rate: float = 1.0        # 1.0C maximum continuous discharge per [D-09]
    emergency_reserve_pct: float = 0.20      # [D-17] Default 20% emergency blackout reserve (configurable in [0.10, 0.50] for BAT-002)
    self_discharge_pct_per_month: float = 2.0  # [D-26] Monthly ambient self-discharge rate (default 2.0%/month for LFP)
    tare_loss_kw: float = 0.0                # [D-26] Standby tare loss in kW (inverter/BMS tare modeled on SoC-side)

            raise ValueError(
                f"round_trip_efficiency must be in (0, 1], got {self.round_trip_efficiency}"
            )
        if not (0.0 < self.usable_dod <= 1.0):
            raise ValueError(f"usable_dod must be in (0, 1], got {self.usable_dod}")
        if not (0.0 <= self.min_soc_pct < 1.0):
            raise ValueError(f"min_soc_pct must be in [0, 1), got {self.min_soc_pct}")
        if not (0.0 <= self.emergency_reserve_pct <= 1.0):
            raise ValueError(
                f"emergency_reserve_pct must be in [0, 1], got {self.emergency_reserve_pct}"
            )
        if self.self_discharge_pct_per_month < 0.0:
            raise ValueError(
                f"self_discharge_pct_per_month must be non-negative, got {self.self_discharge_pct_per_month}"

    """Discrete utility grid outage event per decision [D-11]."""

    start_hour: int
    duration_hours: float
    outage_type: OutageType = OutageType.NORMAL
    solar_depression_factor: float = 1.0
    solar_depression_duration_hours: float = 0.0

    def __post_init__(self) -> None:
        if self.start_hour < 1 or self.start_hour > 8760:
            raise ValueError(f"start_hour must be in [1, 8760], got {self.start_hour}")
        if self.duration_hours <= 0.0:
            raise ValueError(f"duration_hours must be positive, got {self.duration_hours}")
        if not (0.0 <= self.solar_depression_factor <= 1.0):
            raise ValueError(
                f"solar_depression_factor must be in [0, 1], got {self.solar_depression_factor}"
            )

    @property
    def end_hour(self) -> float:
        """Hour at which grid supply is restored."""
        return self.start_hour + self.duration_hours

    def is_active_at(self, hour: int) -> bool:
        """Check whether the grid outage is active at a given chronological hour."""
        return self.start_hour <= hour < self.end_hour

    def is_solar_depressed_at(self, hour: int) -> bool:
        """Check whether solar depression is active at a given chronological hour."""
        depression_end = self.start_hour + self.solar_depression_duration_hours
        return self.start_hour <= hour < depression_end


@dataclass(frozen=True)
class HourlyRecord:
    """Canonical time-step record representing the complete state of the CCIS microgrid.

    Fields match docs/context/conceptual-model.md §4 verbatim.
    """

    hour: int                     # 1 to 8760 (chronological hour of year)
    timestamp: str                # ISO 8601 string (e.g., "2026-04-15T14:00:00+08:00")
    temp_ambient_c: float         # Ambient dry-bulb temperature (°C)
    humidity_rel_pct: float       # Relative humidity (%)
    ghi_w_m2: float               # Global Horizontal Solar Irradiance (W/m²)
    demand_gross_kwh: float       # Total unconstrained building electrical load (kWh)
    demand_served_kwh: float      # Actual electrical load served after load shedding (kWh)
    solar_generation_kwh: float   # Net AC electrical generation from PV array (kWh)
    solar_to_demand_kwh: float    # Direct solar electricity consumed by building (kWh)
    solar_to_battery_kwh: float   # Excess solar electricity routed to battery charging (kWh)
    solar_curtailed_kwh: float    # Excess generation curtailed or exported to grid (kWh)
    battery_soc_kwh: float        # Battery usable energy stored at end of hour (kWh)
    battery_charge_kwh: float     # Energy transferred into battery during this hour (kWh)
    battery_discharge_kwh: float  # Energy extracted from battery during this hour (kWh)
    grid_import_kwh: float        # Electrical energy imported from INEC utility grid (kWh)
    unserved_load_kwh: float      # Deficit electrical energy unserved during blackout (kWh)
    grid_available: bool          # True = INEC grid UP; False = Outage / Blackout
    active_load_tier: int         # 1 = Life Safety only, 2 = Essential IT, 3 = Normal (all loads)

    def __post_init__(self) -> None:
        if self.hour < 1 or self.hour > 8760:
            raise ValueError(f"hour must be in range 1..8760, got {self.hour}")
        if self.active_load_tier not in (1, 2, 3):
            raise ValueError(f"active_load_tier must be 1, 2, or 3, got {self.active_load_tier}")

    @property
    def energy_balance_residual_kwh(self) -> float:
        """Calculate physical energy balance residual: |Total Energy In - Total Energy Out|.

        Total In = solar_generation_kwh + grid_import_kwh + battery_discharge_kwh
        Total Out = demand_served_kwh + battery_charge_kwh + solar_curtailed_kwh
        """
        energy_in = self.solar_generation_kwh + self.grid_import_kwh + self.battery_discharge_kwh
        energy_out = self.demand_served_kwh + self.battery_charge_kwh + self.solar_curtailed_kwh
        return abs(energy_in - energy_out)

    @property
    def signed_residual_kwh(self) -> float:
        """Calculate signed energy balance residual: Total Energy In - Total Energy Out.

        Total In = solar_generation_kwh + grid_import_kwh + battery_discharge_kwh
        Total Out = demand_served_kwh + battery_charge_kwh + solar_curtailed_kwh
        Signed Residual = Total In - Total Out
        """
        energy_in = self.solar_generation_kwh + self.grid_import_kwh + self.battery_discharge_kwh
        energy_out = self.demand_served_kwh + self.battery_charge_kwh + self.solar_curtailed_kwh
        return energy_in - energy_out

    def is_energy_conserved(self, tolerance: float = 1e-5) -> bool:
        """Verify strict energy balance invariant (residual <= 10^-5 kWh)."""
        return self.energy_balance_residual_kwh <= tolerance

    def to_dict(self) -> dict[str, Any]:
        """Convert record to primitive Python dictionary suitable for JSON/CSV/DataFrame."""
        return asdict(self)

    @classmethod
    def from_dict(cls, data: Mapping[str, Any]) -> HourlyRecord:
        """Construct an HourlyRecord from a dictionary or mapping, with type coercion."""
        return cls(
            hour=int(data["hour"]),
            timestamp=str(data["timestamp"]),
            temp_ambient_c=float(data["temp_ambient_c"]),
            humidity_rel_pct=float(data["humidity_rel_pct"]),
            ghi_w_m2=float(data["ghi_w_m2"]),
            demand_gross_kwh=float(data["demand_gross_kwh"]),
            demand_served_kwh=float(data["demand_served_kwh"]),
            solar_generation_kwh=float(data["solar_generation_kwh"]),
            solar_to_demand_kwh=float(data["solar_to_demand_kwh"]),
            solar_to_battery_kwh=float(data["solar_to_battery_kwh"]),
            solar_curtailed_kwh=float(data["solar_curtailed_kwh"]),
            battery_soc_kwh=float(data["battery_soc_kwh"]),
            battery_charge_kwh=float(data["battery_charge_kwh"]),
            battery_discharge_kwh=float(data["battery_discharge_kwh"]),
            grid_import_kwh=float(data["grid_import_kwh"]),
            unserved_load_kwh=float(data["unserved_load_kwh"]),
            grid_available=bool(data["grid_available"]),
            active_load_tier=int(data["active_load_tier"]),
```

```python
"""Equipment audit domain model and inventory parser for CCIS simulation.

Conforms to specifications locked in:
- docs/prompts/CORE-003-PROMPT.md
- docs/decisions/decision-log.md [D-06] (18 rooms layout), [D-10] (Tiers 1, 2, 3)
- docs/decisions/decision-log.md [D-16] (Equipment audit inventory and load assumptions)
"""

from __future__ import annotations

import csv
from dataclasses import dataclass, field
from pathlib import Path
from typing import Any, Sequence

import pandas as pd


@dataclass(frozen=True)
class EquipmentItem:
    """Individual electrical equipment item entry in the building audit.

    Attributes:
        item: Descriptive name of the electrical equipment.
        room_type: Room archetype category (e.g., 'computer_lab', 'lecture_room',
            'hyflex_smart_room', 'dean_and_secretary', 'cod_office', 'faculty_room',
            'building_common').
        count: Total physical count across all rooms of room_type.
        rated_kw: Rated active electrical power per single unit in kW.
        hours_per_day: Nominal operating hours per weekday (0.0 to 24.0).
        tier: Critical priority load shedding tier per [D-10] (1, 2, or 3).
        duty_cycle: Operating diversity factor / duty cycle during active hours (0.0 to 1.0).
        notes: Audit notes, assumption citations, or standard specifications.
    """

    item: str
    room_type: str
    count: int
    rated_kw: float
    hours_per_day: float
    tier: int
    duty_cycle: float = 1.0
    notes: str = ""

    def __post_init__(self) -> None:
        if self.count <= 0:
            raise ValueError(f"count must be a positive integer, got {self.count}")
        if self.rated_kw < 0.0:
            raise ValueError(f"rated_kw must be non-negative, got {self.rated_kw}")
        if not (0.0 <= self.hours_per_day <= 24.0):
            raise ValueError(f"hours_per_day must be in [0, 24], got {self.hours_per_day}")
        if self.tier not in (1, 2, 3):
            raise ValueError(f"tier must be 1, 2, or 3 per [D-10], got {self.tier}")
        if not (0.0 <= self.duty_cycle <= 1.0):
            raise ValueError(f"duty_cycle must be in [0.0, 1.0], got {self.duty_cycle}")

    @property
    def total_kw(self) -> float:
        """Total nameplate connected electrical capacity for this equipment item (kW)."""
        return self.count * self.rated_kw

    @property
    def nominal_daily_kwh(self) -> float:
        """Nominal daily electrical energy consumption without duty cycle derating (kWh/day)."""
        return self.total_kw * self.hours_per_day

    @property
    def effective_daily_kwh(self) -> float:
        """Effective daily electrical energy consumption accounting for duty cycle (kWh/day)."""
        return self.total_kw * self.hours_per_day * self.duty_cycle

    @property
    def is_cooling(self) -> bool:
        """Determine whether the equipment contributes to sensible/latent cooling load."""
        item_lower = self.item.lower()
        return "ac unit" in item_lower or "split ac" in item_lower or "spot cooling" in item_lower


@dataclass
class EquipmentAudit:
    """Comprehensive building equipment audit inventory.

    Attributes:
        items: List of EquipmentItem records.
        floor_area_m2: Gross floor area of facility in square meters (default 2500.0 m² per [D-06]).
        building_name: Name of facility.
    """

    items: list[EquipmentItem] = field(default_factory=list)
    floor_area_m2: float = 2500.0
    building_name: str = "MMSU College of Computing and Information Sciences"

    def __post_init__(self) -> None:
        if self.floor_area_m2 <= 0.0:
            raise ValueError(f"floor_area_m2 must be positive, got {self.floor_area_m2}")

    def add_item(self, item: EquipmentItem) -> None:
        """Add an equipment item to the audit inventory."""
        self.items.append(item)

    @classmethod
    def load_from_csv(
        cls,
        csv_path: Path | str,
        floor_area_m2: float = 2500.0,
        building_name: str = "MMSU College of Computing and Information Sciences",
    ) -> EquipmentAudit:
        """Load and parse an equipment audit CSV fixture.

        Expected minimum columns: item, room_type, count, rated_kw, hours_per_day, tier.
        Optional columns: duty_cycle, notes.
        """
        path = Path(csv_path)
        if not path.exists():
            raise FileNotFoundError(f"Equipment audit fixture file not found: {path}")

        audit = cls(floor_area_m2=floor_area_m2, building_name=building_name)

        with open(path, mode="r", encoding="utf-8") as f:
            reader = csv.DictReader(f)
            required_cols = {"item", "room_type", "count", "rated_kw", "hours_per_day", "tier"}
            if not reader.fieldnames or not required_cols.issubset(set(reader.fieldnames)):
                raise ValueError(
                    f"CSV missing required columns: {required_cols - set(reader.fieldnames or [])}"
                )

            for row in reader:
                item = EquipmentItem(
                    item=row["item"].strip(),
                    room_type=row["room_type"].strip(),
                    count=int(row["count"]),
                    rated_kw=float(row["rated_kw"]),
                    hours_per_day=float(row["hours_per_day"]),
                    tier=int(row["tier"]),
                    duty_cycle=float(row.get("duty_cycle", 1.0) or 1.0),
                    notes=row.get("notes", "").strip(),
                )
                audit.add_item(item)

        return audit

    def total_connected_kw(self, tier: int | None = None) -> float:
        """Calculate total connected electrical power capacity (kW), optionally filtered by tier."""
        if tier is not None:
            return sum(item.total_kw for item in self.items if item.tier == tier)
        return sum(item.total_kw for item in self.items)

    def total_nominal_daily_kwh(self, tier: int | None = None) -> float:
        """Calculate total nominal daily energy consumption (kWh/day), optionally filtered by tier."""
        if tier is not None:
            return sum(item.nominal_daily_kwh for item in self.items if item.tier == tier)
        return sum(item.nominal_daily_kwh for item in self.items)

    def total_effective_daily_kwh(self, tier: int | None = None) -> float:
        """Calculate total effective daily energy consumption (kWh/day), optionally filtered by tier."""
        if tier is not None:
            return sum(item.effective_daily_kwh for item in self.items if item.tier == tier)
        return sum(item.effective_daily_kwh for item in self.items)

    def tier_breakdown_kw(self) -> dict[int, float]:
        """Return connected power in kW broken down by tier {1: kW, 2: kW, 3: kW}."""
        return {
            1: self.total_connected_kw(tier=1),
            2: self.total_connected_kw(tier=2),
            3: self.total_connected_kw(tier=3),
        }

    def tier_breakdown_pct(self) -> dict[int, float]:
        """Return percentage of connected power broken down by tier."""
        total = self.total_connected_kw()
        if total == 0.0:
            return {1: 0.0, 2: 0.0, 3: 0.0}
        return {t: (kw / total) * 100.0 for t, kw in self.tier_breakdown_kw().items()}

    def by_room_type(self, room_type: str) -> list[EquipmentItem]:
        """Filter audit items belonging to a specific room archetype."""
        return [item for item in self.items if item.room_type == room_type]

    def summary_by_room_type(self) -> dict[str, dict[str, float]]:
        """Return aggregated summary metrics by room archetype."""
        summary: dict[str, dict[str, float]] = {}
        for item in self.items:
            if item.room_type not in summary:
                summary[item.room_type] = {
                    "total_kw": 0.0,
                    "nominal_daily_kwh": 0.0,
                    "effective_daily_kwh": 0.0,
                    "unit_count": 0.0,
                }
            summary[item.room_type]["total_kw"] += item.total_kw
            summary[item.room_type]["nominal_daily_kwh"] += item.nominal_daily_kwh
            summary[item.room_type]["effective_daily_kwh"] += item.effective_daily_kwh
            summary[item.room_type]["unit_count"] += item.count
        return summary

    def to_dataframe(self) -> pd.DataFrame:
```

```python
"""Diurnal non-cooling electrical demand model for CCIS facility.

Aggregates bottom-up lighting, IT networking, office computing, and plug loads
across priority tiers 1 to 3 based on the equipment inventory audit and hourly
occupancy schedules, conforming to:
- docs/prompts/DEM-001-PROMPT.md
- docs/context/phase2-interface-contract.md
- docs/decisions/decision-log.md [D-27]
"""

from __future__ import annotations

import numpy as np

from ccis_sim.core.interfaces import OccupancySchedule, TieredDemand
from ccis_sim.core.models import Building
from ccis_sim.demand.equipment import EquipmentAudit


def compute_non_cooling_demand(
    schedule: OccupancySchedule,
    audit: EquipmentAudit,
    building: Building,
) -> TieredDemand:
    r"""Compute hourly non-cooling electrical demand decomposed across priority tiers 1 to 3.

    Conforms strictly to docs/context/phase2-interface-contract.md:
    `compute_non_cooling_demand(schedule: OccupancySchedule, audit: EquipmentAudit, building: Building) -> TieredDemand`

    Load aggregation logic ([ASSUMPTION / [D-27]]):
    1. Filter out all cooling equipment (`item.is_cooling == False`); cooling response is
       exclusively evaluated in DEM-002.
    2. Classify items into:
       - Continuous 24/7 loads (`hours_per_day >= 24.0`): constant uninterrupted operation
         (e.g., Tier 1 life-safety signs/alarms/CCTV/stairwell; Tier 2 IDF switches and core routers).
       - Occupancy-driven loads (`hours_per_day < 24.0`): scaled linearly with hourly
         `occupancy_fraction` (e.g., PC labs, classroom lighting troffers, faculty workstations).
    3. Calculate total daytime peak non-cooling demand:
       $P_{\text{peak}} = P_{\text{cont, total}} + P_{\text{occ\_peak, total}}$.
    4. Unoccupied Baseload Rule without Double-Counting:
       Target unoccupied non-cooling load is 10% of daytime peak ($0.10 \times P_{\text{peak}}$).
       Continuous 24/7 items already draw $P_{\text{cont, total}}$. The residual standby margin:
       $P_{\text{standby}} = \max(0.0, 0.10 \times P_{\text{peak}} - P_{\text{cont, total}})$.
       This residual represents parasitic standby plug loads (quiescent draw of sleep-state PCs,
       monitors, chargers, and appliances) and is allocated to Tier 3 (sheddable load).
       As occupancy rises ($f \in [0, 1]$), equipment wakes to operational load, and standby
       transitions to active draw: $(1.0 - f) \times P_{\text{standby}}$.
    5. Hourly Tier Composition:
       - Tier 1: $P_{\text{cont}, 1} + f \times P_{\text{occ\_peak}, 1}$
       - Tier 2: $P_{\text{cont}, 2} + f \times P_{\text{occ\_peak}, 2}$
       - Tier 3: $P_{\text{cont}, 3} + f \times P_{\text{occ\_peak}, 3} + (1.0 - f) \times P_{\text{standby}}$

    Args:
        schedule: OccupancySchedule containing hourly occupancy_fraction array.
        audit: EquipmentAudit inventory containing audited building loads.
        building: Building envelope definition.

    Returns:
        TieredDemand instance with tier1_kwh, tier2_kwh, tier3_kwh arrays of length N.
    """
    n = len(schedule)
    if n == 0:
        raise ValueError("OccupancySchedule must not be empty.")

    # Filter for non-cooling electrical equipment
    non_cooling_items = [item for item in audit.items if not item.is_cooling]

    # Pre-accumulate continuous loads and full-occupancy peak loads per tier
    # Units: kW = kWh/h at Δt = 1.0 h
    p_cont = {1: 0.0, 2: 0.0, 3: 0.0}
    p_occ_peak = {1: 0.0, 2: 0.0, 3: 0.0}

    for item in non_cooling_items:
        # Effective kW accounts for duty cycle diversity during operational hours
        effective_kw = item.count * item.rated_kw * item.duty_cycle
        if item.hours_per_day >= 24.0:
            p_cont[item.tier] += effective_kw
        else:
            p_occ_peak[item.tier] += effective_kw

    p_cont_total = p_cont[1] + p_cont[2] + p_cont[3]
    p_occ_total = p_occ_peak[1] + p_occ_peak[2] + p_occ_peak[3]
    p_peak = p_cont_total + p_occ_total

    # Baseload target is 10% of daytime peak non-cooling demand per DEM-001.3
    p_unoccupied_target = 0.10 * p_peak

    # Avoid double-counting: subtract continuous 24/7 loads from 10% baseload target
    p_standby_residual = max(0.0, p_unoccupied_target - p_cont_total)

    f_occ = schedule.occupancy_fraction

    # Vectorized hourly tier load generation
    tier1_arr = np.full(n, p_cont[1], dtype=np.float64)
    if p_occ_peak[1] > 0.0:
        tier1_arr += f_occ * p_occ_peak[1]

    tier2_arr = p_cont[2] + f_occ * p_occ_peak[2]
    tier3_arr = p_cont[3] + f_occ * p_occ_peak[3] + (1.0 - f_occ) * p_standby_residual

    return TieredDemand(
        tier1_kwh=tier1_arr,
        tier2_kwh=tier2_arr,
        tier3_kwh=tier3_arr,
    )
```

```python
Implements the Cooling Degree Day (CDD) sensible response, tropical latent humidity
penalty, inverter modulation bounds, and capacity caps conforming to:
- docs/prompts/DEM-002-PROMPT.md
- docs/context/phase2-interface-contract.md
- docs/decisions/decision-log.md [D-07], [D-10], [D-16], [D-28]
"""

import numpy as np
import pandas as pd

from ccis_sim.core.interfaces import TieredDemand
from ccis_sim.core.models import Building
from ccis_sim.demand.equipment import EquipmentAudit


def compute_cooling_demand(
    weather_df: pd.DataFrame,
    occupancy_fraction: np.ndarray,
    audit: EquipmentAudit,
    building: Building,
    *,
    rh_low: float = 60.0,
    rh_high: float = 80.0,
) -> TieredDemand:
    r"""Compute hourly cooling electrical demand decomposed across priority tiers 1 to 3.

    Conforms strictly to docs/context/phase2-interface-contract.md:
    `compute_cooling_demand(weather_df: pd.DataFrame, occupancy_fraction: np.ndarray, audit: EquipmentAudit, building: Building) -> TieredDemand`

    Cooling model formulation ([ASSUMPTION / [D-28]]):
    1. Sensible Cooling Response ([D-07]):
       Sensible temperature delta above balance-point setpoint:
       $\Delta T_i = \max(0.0, T_{\text{amb}, i} - T_{\text{base}})$, where $T_{\text{base}} = \text{building.t\_base\_cooling\_c} = 24.0^\circ\text{C}$.
       Relative sensitivity: $s_T = \text{building.cooling\_sensitivity\_per\_c} = 0.04$ ($4.0\%/^\circ\text{C}$).
       At or below $T_{\text{base}}$, sensible cooling demand is identically zero.
    2. Tropical Latent Humidity Penalty ([D-07]):
       At tropical high humidity, coil moisture condensation increases enthalpy load:
       $M_{\text{latent}}(RH) = 1.0 + \text{building.latent\_cooling\_penalty} \times \text{clip}\left(\frac{RH - RH_{\text{low}}}{RH_{\text{high}} - RH_{\text{low}}}, 0.0, 1.0\right)$.
       Below $RH_{\text{low}} = 60.0\%$, $M_{\text{latent}} = 1.00$. Above $RH_{\text{high}} = 80.0\%$, $M_{\text{latent}} = 1.20$ (+20% penalty).
       Between 60% and 80%, penalty ramps linearly, avoiding unphysical step discontinuities.
    3. Operational Categories and Tier Allocation ([D-10], [D-16]):
       - Tier 2 Spot Cooling (IDF / telecommunications closets):
         Runs 24/7 uninterrupted ($hours = 24.0$) to dissipate continuous heat from PoE switches
         and routers. Independent of human occupancy ($f_{\text{occ}}$):
         $P_{\text{spot}, i} = \min\left(P_{\text{conn, spot}}, P_{\text{conn, spot}} \times s_T \Delta T_i \times M_{\text{latent}}(RH_i)\right)$.
       - Tier 3 Comfort Cooling (Classrooms, PC labs, faculty/admin offices):
         Operates only when rooms are occupied ($f_{\text{occ}} > 0$). When unoccupied ($f_{\text{occ}} = 0$),
         comfort AC units are turned off:
         $P_{\text{comfort}, i} = \min\left(f_{\text{occ}, i} \times P_{\text{conn, comfort}}, f_{\text{occ}, i} \times P_{\text{conn, comfort}} \times s_T \Delta T_i \times M_{\text{latent}}(RH_i)\right)$.
       - Tier 1 Life-Safety:
         Zero cooling equipment is assigned to Tier 1 in the building audit ($P_{\text{Tier 1}} = 0.0$).
    4. Capacity Cap and Modulation Floor:
       Cooling electrical demand is hard-capped at connected cooling capacity per tier.
       Inverter compressors modulate continuously; cycling at minimum speed over the 1-hour interval
       yields smooth energy integration proportional to required thermal load.

    Args:
        weather_df: DataFrame containing 't_amb' (°C) and 'rh' (%) time series.
        occupancy_fraction: 1D numpy array of length N containing float values in [0.0, 1.0].
        audit: EquipmentAudit inventory containing audited equipment items.
        building: Building envelope providing t_base_cooling_c, cooling_sensitivity_per_c,
            and latent_cooling_penalty.
        rh_low: Lower relative humidity threshold for latent penalty ramp (default 60.0%).
        rh_high: Upper relative humidity threshold for full latent penalty (default 80.0%).

    Returns:
        TieredDemand instance containing tier1_kwh, tier2_kwh, and tier3_kwh arrays of length N.
    """
    n = len(weather_df)
    occ = np.asarray(occupancy_fraction, dtype=np.float64)
    if len(occ) != n:
        raise ValueError(
            f"occupancy_fraction length ({len(occ)}) must match weather_df length ({n})."
        )
    if n == 0:
        raise ValueError("weather_df must not be empty.")

    if "t_amb" not in weather_df.columns or "rh" not in weather_df.columns:
        raise ValueError("weather_df must contain 't_amb' and 'rh' columns.")

    t_amb = weather_df["t_amb"].to_numpy(dtype=np.float64)
    rh = weather_df["rh"].to_numpy(dtype=np.float64)

    # 1. Delta-T above base temperature
    dt = np.maximum(0.0, t_amb - building.t_base_cooling_c)

    # 2. Continuous piecewise-linear latent humidity multiplier
    rh_norm = np.clip((rh - rh_low) / (rh_high - rh_low), 0.0, 1.0)
    m_latent = 1.0 + building.latent_cooling_penalty * rh_norm

    # 3. Disaggregate cooling equipment from audit by tier and operational schedule
    cooling_items = [it for it in audit.items if it.is_cooling]

    # Pre-accumulate connected capacity for spot cooling (24/7) vs comfort cooling per tier
    # Spot cooling: hours_per_day >= 24.0; Comfort cooling: hours_per_day < 24.0
    p_conn_spot = {1: 0.0, 2: 0.0, 3: 0.0}
    p_conn_comfort = {1: 0.0, 2: 0.0, 3: 0.0}

    for item in cooling_items:
        conn_kw = item.count * item.rated_kw
        if item.hours_per_day >= 24.0:
            p_conn_spot[item.tier] += conn_kw
        else:
            p_conn_comfort[item.tier] += conn_kw

    s_t = building.cooling_sensitivity_per_c

    # Vectorized hourly cooling calculation per tier
    tier_arrays: dict[int, np.ndarray] = {}
    for tier in (1, 2, 3):
        tier_arr = np.zeros(n, dtype=np.float64)

        # Spot cooling term (independent of human occupancy, 24/7)
        if p_conn_spot[tier] > 0.0:
            spot_demand = p_conn_spot[tier] * s_t * dt * m_latent
            tier_arr += np.minimum(p_conn_spot[tier], spot_demand)

        # Comfort cooling term (scaled by occupancy_fraction)
        if p_conn_comfort[tier] > 0.0:
            comfort_cap = occ * p_conn_comfort[tier]
            comfort_demand = occ * p_conn_comfort[tier] * s_t * dt * m_latent
            tier_arr += np.minimum(comfort_cap, comfort_demand)

        tier_arrays[tier] = tier_arr

    return TieredDemand(
        tier1_kwh=tier_arrays[1],
        tier2_kwh=tier_arrays[2],
        tier3_kwh=tier_arrays[3],
    )
```

```python
"""Solar irradiance geometry and NOCT cell temperature thermal derating model.

Calculates hourly plane-of-array (POA) irradiance for a tilted rooftop PV array
at Batac, Ilocos Norte, models cell operating temperature with rooftop mounting
thermal penalty, and applies temperature derating to compute DC power yield per kWp.

Conforms to Phase 2 interface contract (docs/context/phase2-interface-contract.md)
and decisions [D-08] and [D-29].
"""

from __future__ import annotations

import logging
from typing import NamedTuple

import numpy as np
import pandas as pd

logger = logging.getLogger(__name__)

# Standard constants
SOLAR_CONSTANT_W_M2 = 1367.0  # W/m^2 solar constant (G_sc)
STC_IRRADIANCE_W_M2 = 1000.0  # W/m^2 Standard Test Conditions (STC) irradiance
STC_CELL_TEMP_C = 25.0  # °C STC cell reference temperature

# Physical modeling parameters per [D-08] and [D-29]
DEFAULT_TILT_DEG = 15.0  # degrees South-facing optimal tilt for Batac (~18.06°N)
DEFAULT_AZIMUTH_DEG = 180.0  # degrees True South (North = 0°, South = 180°)
BATAC_LATITUDE_DEG = 18.06  # degrees North
BATAC_LONGITUDE_DEG = 120.56  # degrees East
STANDARD_MERIDIAN_DEG = 120.0  # degrees East (UTC+8 Philippine Standard Time)

# [ASSUMPTION] Ground diffuse albedo for academic campus surroundings (grass/concrete mix) per [D-29]
DEFAULT_ALBEDO = 0.20

# [ASSUMPTION] Nominal Operating Cell Temperature with +10°C rooftop thermal penalty per [D-08] and [D-29]
DEFAULT_NOCT_C = 45.0  # °C baseline manufacturer NOCT
ROOFTOP_PENALTY_C = 10.0  # °C close-roof mounting thermal penalty per [D-08]
EFFECTIVE_NOCT_C = DEFAULT_NOCT_C + ROOFTOP_PENALTY_C  # 55.0 °C effective NOCT

# Temperature coefficient of power per [D-08]
TEMP_COEFFICIENT_PER_C = -0.0040  # -0.40%/°C for monocrystalline silicon


class SolarPosition(NamedTuple):
    """Astronomical solar position and geometric angles."""

    solar_time_hours: np.ndarray
    hour_angle_rad: np.ndarray
    declination_rad: np.ndarray
    cos_zenith: np.ndarray
    zenith_rad: np.ndarray
    elevation_deg: np.ndarray
    cos_incidence: np.ndarray


def compute_solar_position(
    timestamps: pd.Series | pd.DatetimeIndex | np.ndarray,
    *,
    tilt_deg: float = DEFAULT_TILT_DEG,
    azimuth_deg: float = DEFAULT_AZIMUTH_DEG,
    latitude_deg: float = BATAC_LATITUDE_DEG,
    longitude_deg: float = BATAC_LONGITUDE_DEG,
    standard_meridian_deg: float = STANDARD_MERIDIAN_DEG,
) -> SolarPosition:
    """Compute astronomical solar position and surface incidence angle.

    Uses the Spencer (1971) Fourier series formulation for solar declination
    and Equation of Time (EoT), with longitude correction for local solar time.

    Args:
        timestamps: Series or array of datetimes (UTC+8 / PST).
        tilt_deg: Surface tilt angle from horizontal in degrees (default: 15.0°).
        azimuth_deg: Surface azimuth from True North in degrees (default: 180.0° / South).
        latitude_deg: Site latitude in degrees North (default: 18.06°).
        longitude_deg: Site longitude in degrees East (default: 120.56°).
        standard_meridian_deg: Timezone standard meridian in degrees East (default: 120.0°).

    Returns:
        SolarPosition named tuple containing solar time, hour angle, declination,
        cos(zenith), zenith, elevation, and cos(incidence).
    """
    dti = pd.DatetimeIndex(pd.to_datetime(timestamps))
    day_of_year = dti.dayofyear.values.astype(np.float64)
    local_hours = (
        dti.hour.values.astype(np.float64)
        + dti.minute.values.astype(np.float64) / 60.0
        + dti.second.values.astype(np.float64) / 3600.0
    )

    phi = np.radians(latitude_deg)
    beta = np.radians(tilt_deg)
    # Duffie & Beckman surface azimuth from South: gamma_DB = azimuth - 180°
    gamma_db = np.radians(azimuth_deg - 180.0)

    # Fractional day angle B in radians (Spencer 1971)
    b = 2.0 * np.pi * (day_of_year - 1.0) / 365.0

    # Spencer Fourier series for solar declination (radians)
    delta = (
        0.006918
        - 0.399912 * np.cos(b)
        + 0.070257 * np.sin(b)
        - 0.006758 * np.cos(2.0 * b)
        + 0.000907 * np.sin(2.0 * b)
        - 0.002697 * np.cos(3.0 * b)
        + 0.00148 * np.sin(3.0 * b)
    )

    # Spencer Equation of Time (minutes)
    eot_min = 229.18 * (
        0.000075
        + 0.001868 * np.cos(b)
        - 0.032077 * np.sin(b)
        - 0.014615 * np.cos(2.0 * b)
        - 0.040849 * np.sin(2.0 * b)
    )

    # Longitude time correction (minutes): 4 min per degree East of meridian
    lon_correction_min = 4.0 * (longitude_deg - standard_meridian_deg)

    # Local Solar Time (hours)
    solar_time = local_hours + (eot_min + lon_correction_min) / 60.0

    # Solar hour angle omega (radians, omega = 0 at solar noon)
    omega = np.radians(15.0 * (solar_time - 12.0))

    # Solar zenith angle: cos(theta_z)
    cos_theta_z = np.sin(phi) * np.sin(delta) + np.cos(phi) * np.cos(delta) * np.cos(omega)
    cos_theta_z = np.clip(cos_theta_z, -1.0, 1.0)
    theta_z = np.arccos(cos_theta_z)
    elevation_deg = 90.0 - np.degrees(theta_z)

    # Angle of incidence theta on tilted plane (Duffie & Beckman eq 1.6.2):
    # cos(theta) = sin(delta)*sin(phi)*cos(beta) - sin(delta)*cos(phi)*sin(beta)*cos(gamma_DB)
    #              + cos(delta)*cos(phi)*cos(beta)*cos(omega)
    #              + cos(delta)*sin(phi)*sin(beta)*cos(gamma_DB)*cos(omega)
    #              + cos(delta)*sin(beta)*sin(gamma_DB)*sin(omega)
    cos_incidence = (
        np.sin(delta) * np.sin(phi) * np.cos(beta)
        - np.sin(delta) * np.cos(phi) * np.sin(beta) * np.cos(gamma_db)
        + np.cos(delta) * np.cos(phi) * np.cos(beta) * np.cos(omega)
        + np.cos(delta) * np.sin(phi) * np.sin(beta) * np.cos(gamma_db) * np.cos(omega)
        + np.cos(delta) * np.sin(beta) * np.sin(gamma_db) * np.sin(omega)
    )
    cos_incidence = np.clip(cos_incidence, -1.0, 1.0)

    return SolarPosition(
        solar_time_hours=solar_time,
        hour_angle_rad=omega,
        declination_rad=delta,
        cos_zenith=cos_theta_z,
        zenith_rad=theta_z,
        elevation_deg=elevation_deg,
        cos_incidence=cos_incidence,
    )


def decompose_ghi_erbs(
    ghi_w_m2: np.ndarray,
    cos_zenith: np.ndarray,
    day_of_year: np.ndarray,
) -> tuple[np.ndarray, np.ndarray]:
    """Decompose Global Horizontal Irradiance into DHI and DNI via the Erbs model.

    Implements the Erbs, Klein, and Duffie (1982) empirical correlation relating
    the diffuse fraction kd = DHI / GHI to the hourly clearness index kt = GHI / G_0.

    Args:
        ghi_w_m2: 1D array of global horizontal irradiance (W/m^2).
        cos_zenith: 1D array of cos(theta_z) solar zenith cosines.
        day_of_year: 1D array of day numbers (1 to 365/366).

    Returns:
        Tuple of (dhi_w_m2, dni_w_m2) 1D numpy arrays.
    """
    ghi = np.asarray(ghi_w_m2, dtype=np.float64)
    cos_z = np.asarray(cos_zenith, dtype=np.float64)
    doy = np.asarray(day_of_year, dtype=np.float64)

    # Extraterrestrial normal and horizontal irradiance
    # G_on = G_sc * [1 + 0.033 * cos(2*pi*n/365)]
    g_on = SOLAR_CONSTANT_W_M2 * (1.0 + 0.033 * np.cos(2.0 * np.pi * doy / 365.0))
    g_0 = np.maximum(0.0, g_on * cos_z)

    # Daytime mask: sun above horizon and positive irradiance
    daytime = (ghi > 0.0) & (cos_z > 1e-3)

    # Hourly clearness index kt
    kt = np.zeros_like(ghi)
    safe_g0 = np.maximum(g_0[daytime], 1.0)
    kt[daytime] = np.clip(ghi[daytime] / safe_g0, 0.0, 1.2)

    # Erbs piecewise correlation for diffuse fraction kd
    kd = np.ones_like(ghi)  # Night or zero irradiance defaults to 1.0

    mask1 = daytime & (kt <= 0.22)
    kd[mask1] = 1.0 - 0.09 * kt[mask1]

    mask2 = daytime & (kt > 0.22) & (kt <= 0.80)
    kt2 = kt[mask2]
    kd[mask2] = 0.9511 - 0.1604 * kt2 + 4.388 * (kt2**2) - 16.638 * (kt2**3) + 12.336 * (kt2**4)

    mask3 = daytime & (kt > 0.80)
    kd[mask3] = 0.165

    # Compute DHI and DNI
    dhi = np.zeros_like(ghi)
    dhi[daytime] = np.clip(ghi[daytime] * kd[daytime], 0.0, ghi[daytime])

    dni = np.zeros_like(ghi)
    dni[daytime] = np.maximum(0.0, (ghi[daytime] - dhi[daytime]) / cos_z[daytime])
    # Cap DNI at extraterrestrial normal radiation
    dni[daytime] = np.minimum(dni[daytime], g_on[daytime])

    return dhi, dni


def transpose_irradiance_isotropic(
    ghi_w_m2: np.ndarray,
    dhi_w_m2: np.ndarray,
    dni_w_m2: np.ndarray,
    cos_zenith: np.ndarray,
    cos_incidence: np.ndarray,
    *,
    tilt_deg: float = DEFAULT_TILT_DEG,
    albedo: float = DEFAULT_ALBEDO,
) -> np.ndarray:
    """Transpose horizontal solar components to plane-of-array (POA) irradiance.

    Uses the Liu & Jordan (1963) isotropic sky diffuse model:
    G_poa = G_beam + G_diffuse_sky + G_ground_reflected

    Args:
        ghi_w_m2: Global horizontal irradiance (W/m^2).
        dhi_w_m2: Diffuse horizontal irradiance (W/m^2).
        dni_w_m2: Direct normal irradiance (W/m^2).
        cos_zenith: Cosine of solar zenith angle.
        cos_incidence: Cosine of surface incidence angle.
        tilt_deg: Surface tilt from horizontal (degrees).
        albedo: Ground reflectance factor (default: 0.20 per [D-29]).

    Returns:
        1D numpy array of total POA irradiance (W/m^2).
    """
    ghi = np.asarray(ghi_w_m2, dtype=np.float64)
    dhi = np.asarray(dhi_w_m2, dtype=np.float64)
    dni = np.asarray(dni_w_m2, dtype=np.float64)
    cos_z = np.asarray(cos_zenith, dtype=np.float64)
    cos_inc = np.asarray(cos_incidence, dtype=np.float64)

    beta = np.radians(tilt_deg)
    daytime = (ghi > 0.0) & (cos_z > 1e-3)

    # 1. Beam on tilted plane: DNI * max(0, cos(theta))
    g_beam = np.zeros_like(ghi)
    g_beam[daytime] = dni[daytime] * np.maximum(0.0, cos_inc[daytime])

    # 2. Isotropic sky diffuse: DHI * (1 + cos(beta)) / 2
    g_sky = np.zeros_like(ghi)
    g_sky[daytime] = dhi[daytime] * (1.0 + np.cos(beta)) / 2.0

    # 3. Ground-reflected diffuse: GHI * rho_g * (1 - cos(beta)) / 2
    g_ground = np.zeros_like(ghi)
    g_ground[daytime] = ghi[daytime] * albedo * (1.0 - np.cos(beta)) / 2.0

    g_poa = np.zeros_like(ghi)
    g_poa[daytime] = g_beam[daytime] + g_sky[daytime] + g_ground[daytime]
    g_poa = np.maximum(0.0, g_poa)

    return g_poa


def compute_cell_temperature(
    t_amb_c: np.ndarray,
    g_poa_w_m2: np.ndarray,
    *,
    effective_noct_c: float = EFFECTIVE_NOCT_C,
) -> np.ndarray:
    """Compute PV cell operating temperature including rooftop mounting penalty.

    Formula per [D-08] and [D-29]:
    T_cell = T_amb + ((effective_noct - 20.0) / 800.0) * G_poa
    where effective_noct = NOCT_base (45°C) + rooftop_penalty (+10°C) = 55°C.

    Args:
        t_amb_c: Ambient dry-bulb temperature (°C).
        g_poa_w_m2: Plane-of-array irradiance (W/m^2).
        effective_noct_c: Effective NOCT accounting for mounting thermal penalty (°C).

    Returns:
        1D numpy array of cell operating temperature (°C).
    """
    t_amb = np.asarray(t_amb_c, dtype=np.float64)
    g_poa = np.asarray(g_poa_w_m2, dtype=np.float64)

    # (55.0 - 20.0) / 800.0 = 35.0 / 800.0 = 0.04375 °C / (W/m^2)
    thermal_slope = (effective_noct_c - 20.0) / 800.0
    t_cell = t_amb + thermal_slope * np.maximum(0.0, g_poa)
    return t_cell


def compute_thermal_derating_factor(
    t_cell_c: np.ndarray,
    *,
    gamma_per_c: float = TEMP_COEFFICIENT_PER_C,
    t_ref_c: float = STC_CELL_TEMP_C,
) -> np.ndarray:
    """Compute module temperature derating factor relative to STC (25°C).

    Formula per [D-08]:
    f_temp = max(0.0, 1.0 + gamma * (T_cell - 25.0))
    where gamma = -0.40%/°C (-0.0040/°C).

    Args:
        t_cell_c: Cell operating temperature (°C).
        gamma_per_c: Temperature coefficient of power (1/°C, negative).
        t_ref_c: Reference cell temperature (°C, default: 25.0).

    Returns:
        1D numpy array of dimensionless thermal derating factors.
    """
    t_cell = np.asarray(t_cell_c, dtype=np.float64)
    f_temp = 1.0 + gamma_per_c * (t_cell - t_ref_c)
    return np.maximum(0.0, f_temp)


def compute_dc_power_per_kwp(
    weather_df: pd.DataFrame,
    *,
    tilt_deg: float = DEFAULT_TILT_DEG,
    azimuth_deg: float = DEFAULT_AZIMUTH_DEG,
    latitude_deg: float = BATAC_LATITUDE_DEG,
    longitude_deg: float = BATAC_LONGITUDE_DEG,
) -> np.ndarray:
    """Compute hourly DC power output per kWp for a rooftop PV array at Batac.

    Transforms hourly Global Horizontal Irradiance (GHI) and ambient temperature
    into plane-of-array irradiance, calculates cell operating temperature under
    rooftop thermal mounting penalties, and applies module thermal derating.

    Outputs DC power per kWp BEFORE balance-of-system and inverter losses
    (which are modeled in SOL-002 per docs/context/phase2-interface-contract.md).

    Args:
        weather_df: Verified weather DataFrame containing columns:
            ['hour', 'timestamp', 't_amb', 'rh', 'ghi'].
        tilt_deg: Array tilt angle from horizontal in degrees (default: 15.0°).
        azimuth_deg: Array azimuth angle from North in degrees (default: 180.0° / South).
        latitude_deg: Site latitude in degrees North (default: 18.06°).
        longitude_deg: Site longitude in degrees East (default: 120.56°).

    Returns:
        1D numpy float array of length N (8,760 for annual runs) representing
        hourly DC power generation in kW DC per kWp (equivalent to kWh/kWp at Δt = 1.0h).
    """
    if "ghi" not in weather_df.columns:
        raise ValueError("Weather DataFrame missing required column 'ghi'.")
    if "t_amb" not in weather_df.columns:
        raise ValueError("Weather DataFrame missing required column 't_amb'.")

    n_rows = len(weather_df)
    if n_rows == 0:
        return np.empty(0, dtype=np.float64)

    ghi = weather_df["ghi"].values.astype(np.float64)
    t_amb = weather_df["t_amb"].values.astype(np.float64)

    # Validate inputs
    if not np.all(np.isfinite(ghi)):
        raise ValueError("Column 'ghi' contains non-finite values (NaN or inf).")
    if not np.all(np.isfinite(t_amb)):
        raise ValueError("Column 't_amb' contains non-finite values (NaN or inf).")
    if np.any(ghi < 0.0):
        raise ValueError("Column 'ghi' contains negative irradiance values.")

    # Determine timestamps / days of year
    if "timestamp" in weather_df.columns:
        ts = pd.DatetimeIndex(pd.to_datetime(weather_df["timestamp"]))
        doy = ts.dayofyear.values.astype(np.float64)
    elif "hour" in weather_df.columns:
        # Fallback if only 'hour' is provided (1..8760 standard annual sequence)
        hours = weather_df["hour"].values.astype(int)
        # Hour 1 corresponds to 2019-01-01 00:00:00+08:00
        ref_start = pd.Timestamp("2019-01-01 00:00:00+08:00")
        ts = pd.date_range(start=ref_start, periods=len(hours), freq="1h")
        doy = ts.dayofyear.values.astype(np.float64)
    else:
        raise ValueError("Weather DataFrame must contain either 'timestamp' or 'hour' column.")

    # 1. Astronomical solar position
    pos = compute_solar_position(
        ts,
        tilt_deg=tilt_deg,
        azimuth_deg=azimuth_deg,
        latitude_deg=latitude_deg,
        longitude_deg=longitude_deg,
    )

    # 2. GHI -> DHI / DNI decomposition (Erbs correlation)
    dhi, dni = decompose_ghi_erbs(ghi, pos.cos_zenith, doy)

    # 3. Transposition to Plane of Array (Liu-Jordan Isotropic)
    g_poa = transpose_irradiance_isotropic(
        ghi,
        dhi,
        dni,
        pos.cos_zenith,
        pos.cos_incidence,
        tilt_deg=tilt_deg,
        albedo=DEFAULT_ALBEDO,
    )

    # 4. Cell temperature calculation (NOCT + rooftop thermal penalty)
    t_cell = compute_cell_temperature(t_amb, g_poa, effective_noct_c=EFFECTIVE_NOCT_C)

    # 5. Thermal derating factor relative to STC
    f_temp = compute_thermal_derating_factor(t_cell, gamma_per_c=TEMP_COEFFICIENT_PER_C)

    # 6. DC Power output per kWp: (G_poa / 1000 W/m^2) * f_temp
    # Strictly zero whenever GHI is zero or sun is below horizon
    daytime = (ghi > 0.0) & (pos.cos_zenith > 1e-3)
    dc_power_kw_per_kwp = np.zeros(n_rows, dtype=np.float64)
    dc_power_kw_per_kwp[daytime] = (g_poa[daytime] / STC_IRRADIANCE_W_M2) * f_temp[daytime]

    # Clean numerical precision
    dc_power_kw_per_kwp = np.maximum(0.0, dc_power_kw_per_kwp)

    return dc_power_kw_per_kwp
```

```python
"""Solar PV array capacity scaling, inverter clipping, and balance-of-system losses.

Implements conversion of hourly DC generation per kWp (from SOL-001) into AC power

# [ASSUMPTION] Peak inverter efficiency per [D-08] and [D-30]

# [ASSUMPTION] Named balance-of-system (BOS) derate factors per [D-08] and [D-30]

    *,

) -> float:
    """Calculate the inverter rated continuous AC power capacity (kW AC).

    Formula per [D-30]:
    P_ac_rated = capacity_kwp / dc_ac_ratio

    Args:
        capacity_kwp: Nameplate DC capacity of the PV array (kWp).
        dc_ac_ratio: Inverter loading ratio (default: 1.20).

    Returns:

    """

def compute_ac_generation(
    dc_kw_per_kwp: np.ndarray,
    array: SolarArray,
    *,
    inverter_efficiency: float = DEFAULT_INVERTER_EFFICIENCY,
    dc_ac_ratio: float = DEFAULT_DC_AC_RATIO,
    soiling: float = DEFAULT_SOILING_FACTOR,
    dc_wiring: float = DEFAULT_DC_WIRING_FACTOR,
    ac_wiring: float = DEFAULT_AC_WIRING_FACTOR,
    mismatch: float = DEFAULT_MISMATCH_FACTOR,
    availability: float = DEFAULT_AVAILABILITY_FACTOR,
) -> np.ndarray:
    """Compute hourly AC power generation for a sized solar PV array.

    Scales array capacity, models inverter peak efficiency and DC/AC clipping limit,
    and applies non-thermal balance-of-system (BOS) loss factors to produce hourly
    AC power output. Does not double-count temperature losses, as temperature derating
    is explicitly modeled upstream in dc_kw_per_kwp (SOL-001).

    Conforms to Phase 2 interface contract (docs/context/phase2-interface-contract.md).

    Args:
        dc_kw_per_kwp: 1D numpy array of length N (8,760) representing hourly DC yield
            per unit capacity (kW DC / kWp), computed by SOL-001.
        array: SolarArray instance specifying capacity_kwp and parameters.
        inverter_efficiency: Inverter peak conversion efficiency (default: 0.98 per [D-30]).
        dc_ac_ratio: Inverter DC/AC sizing ratio for clipping (default: 1.20 per [D-30]).
        soiling: Soiling derating factor (default: 0.96).
        dc_wiring: DC wiring derating factor (default: 0.985).
        ac_wiring: AC wiring derating factor (default: 0.99).
        mismatch: Module mismatch derating factor (default: 0.98).
        availability: System availability derating factor (default: 0.99).

    Returns:
        1D numpy float64 array of length N representing hourly AC energy output in kWh/h
        (numerically equivalent to kW AC at Δt = 1.0 h). Guaranteed strictly non-negative,
        bounded by the inverter AC rating, and 0.0 at night and for 0 kWp arrays.
    """
    dc_in = np.asarray(dc_kw_per_kwp, dtype=np.float64)

    if len(dc_in) == 0:
        return np.empty(0, dtype=np.float64)

    if not np.all(np.isfinite(dc_in)):
        raise ValueError("Input 'dc_kw_per_kwp' contains non-finite values (NaN or inf).")
    if np.any(dc_in < 0.0):
        raise ValueError("Input 'dc_kw_per_kwp' contains negative values.")

    # Capacity scenario 0 kWp returns strict zero array
    if array.capacity_kwp == 0.0:
        return np.zeros_like(dc_in)

    # 1. Scale DC power to nameplate capacity
    # dc_raw in kW DC
    dc_raw = dc_in * array.capacity_kwp

    # 2. Apply DC-side balance-of-system losses (soiling, DC wiring, mismatch, availability)
    dc_bos_factor = soiling * dc_wiring * mismatch * availability
    dc_at_inverter = dc_raw * dc_bos_factor

    # 3. Inverter conversion and clipping
    # Gross AC power before clipping (kW)
    ac_gross = dc_at_inverter * inverter_efficiency

    # Inverter AC rating limit (kW AC)
    ac_rating = get_inverter_ac_rating(array.capacity_kwp, dc_ac_ratio=dc_ac_ratio)

    # Inverter clipping: AC output cannot exceed continuous rating
    ac_clipped = np.minimum(ac_gross, ac_rating)

    # 4. Apply AC-side collection and wiring loss
    ac_net = ac_clipped * ac_wiring

    # Clean numerical precision and guarantee strict non-negativity
    ac_net = np.maximum(0.0, ac_net)

    return ac_net
```

```python
"""Battery dispatch dynamics, State-of-Charge tracking, and physical limits.

Implements the single time-step battery dispatch algorithm (BAT-001) per
decision [D-25], [D-26], and [D-31].
"""

from __future__ import annotations

import math

from ccis_sim.core.interfaces import BatteryStepResult
from ccis_sim.core.models import Battery


def step_battery(
    soc_kwh: float,
    battery: Battery,
    charge_request_kwh: float,
    discharge_request_kwh: float,
    *,
    soc_floor_kwh: float | None = None,
    delta_t_hours: float = 1.0,
) -> BatteryStepResult:
    """Dispatches battery energy storage for a single discrete time-step.

    Computes AC-side charge and discharge flows, shortfalls, and the resulting
    chemical State-of-Charge (SoC) subject to physical inverter power limits
    (C-rate), usable energy boundaries, and symmetric conversion efficiency.

    [ASSUMPTION] Per decision [D-25], charge and discharge are modeled strictly
    as AC-side flows (matching microgrid bus power conservation). Electrochemical
    and power electronics conversion losses are accounted for symmetrically on the
    chemical State-of-Charge:
        eta_chg = eta_dis = sqrt(battery.round_trip_efficiency).
    When charging:
        Delta SoC = +eta_chg * charge_kwh
    When discharging:
        Delta SoC = -discharge_kwh / eta_dis

    [ASSUMPTION] Idle losses, ambient self-discharge (2%/month), and inverter tare
    standby losses are explicitly excluded from this module per [D-26] and are
    managed independently by BAT-002 (ccis_sim.battery.reserve.apply_idle_losses).

    [ASSUMPTION] Simultaneous charge and discharge requests exceeding 1e-6 kWh
    each are physically mutually exclusive and raise a ValueError at the source.

    [ASSUMPTION] Floating-point comparison tolerance fp_tol = 1e-9 is applied
    to guard boundary evaluations against benign numerical representation noise.

    Args:
        soc_kwh: Stored chemical State-of-Charge at the start of the timestep (kWh).
        battery: Battery specification defining capacity, C-rates, and efficiency.
        charge_request_kwh: Desired AC-side energy to charge into the battery (kWh).
        discharge_request_kwh: Desired AC-side energy to discharge from the battery (kWh).
        soc_floor_kwh: Optional minimum allowable chemical SoC floor (kWh).
            Defaults to battery.min_soc_kwh if None.
        delta_t_hours: Timestep duration in hours (default 1.0).

    Returns:
        BatteryStepResult containing resulting chemical SoC, AC-side flows, and shortfalls.

    Raises:
        TypeError: If battery is not a Battery instance or numeric fields are invalid types.
        ValueError: If non-finite values are passed, requests are negative, delta_t_hours <= 0,
            or simultaneous charging and discharging is requested.
    """
    if not isinstance(battery, Battery):
        raise TypeError(f"battery must be an instance of Battery, got {type(battery)}")

    # Type and finiteness validation
    for name, val in (
        ("soc_kwh", soc_kwh),
        ("charge_request_kwh", charge_request_kwh),
        ("discharge_request_kwh", discharge_request_kwh),
        ("delta_t_hours", delta_t_hours),
    ):
        if not isinstance(val, (int, float)):
            raise TypeError(f"'{name}' must be numeric, got {type(val)}")
        if math.isnan(val) or math.isinf(val):
            raise ValueError(f"'{name}' must be finite, got {val}")

    if delta_t_hours <= 0.0:
        raise ValueError(f"delta_t_hours must be strictly positive, got {delta_t_hours}")

    if charge_request_kwh < 0.0:
        raise ValueError(f"charge_request_kwh must be non-negative, got {charge_request_kwh}")

    if discharge_request_kwh < 0.0:
        raise ValueError(f"discharge_request_kwh must be non-negative, got {discharge_request_kwh}")

    fp_tol = 1e-9

    # Physical boundary check on initial SoC
    if soc_kwh < -fp_tol:
        raise ValueError(f"soc_kwh cannot be physically negative, got {soc_kwh}")

    # Mutual exclusivity check
    if charge_request_kwh > 1e-6 and discharge_request_kwh > 1e-6:
        raise ValueError(
            f"Simultaneous charging ({charge_request_kwh:.6f} kWh) and discharging "
            f"({discharge_request_kwh:.6f} kWh) is physically prohibited."
        )

    # Zero-capacity battery edge case (e.g. grid-only baseline per [D-23])
    if battery.capacity_kwh == 0.0:
        return BatteryStepResult(
            soc_kwh=0.0,
            charge_kwh=0.0,
            discharge_kwh=0.0,
            charge_curtailed_kwh=charge_request_kwh,
            discharge_unmet_kwh=discharge_request_kwh,
        )

    if soc_kwh > battery.capacity_kwh + fp_tol:
        raise ValueError(
            f"soc_kwh ({soc_kwh:.6f} kWh) exceeds battery rated capacity "
            f"({battery.capacity_kwh:.6f} kWh)."
        )

    # Validate soc_floor_kwh if provided
    if soc_floor_kwh is not None:
        if not isinstance(soc_floor_kwh, (int, float)):
            raise TypeError(f"'soc_floor_kwh' must be numeric, got {type(soc_floor_kwh)}")
        if math.isnan(soc_floor_kwh) or math.isinf(soc_floor_kwh):
            raise ValueError(f"'soc_floor_kwh' must be finite, got {soc_floor_kwh}")
        if soc_floor_kwh < -fp_tol:
            raise ValueError(f"'soc_floor_kwh' cannot be physically negative, got {soc_floor_kwh}")
        if soc_floor_kwh > battery.capacity_kwh + fp_tol:
            raise ValueError(
                f"'soc_floor_kwh' ({soc_floor_kwh:.6f} kWh) cannot exceed battery capacity "
                f"({battery.capacity_kwh:.6f} kWh)."
            )
        effective_floor = max(0.0, min(battery.capacity_kwh, float(soc_floor_kwh)))
    else:
        effective_floor = battery.min_soc_kwh

    # Clamp initial SoC within [0, capacity] to absorb fp noise
    cur_soc = max(0.0, min(battery.capacity_kwh, float(soc_kwh)))

    # Symmetric efficiency split per [D-25]
    eta_chg = math.sqrt(battery.round_trip_efficiency)
    eta_dis = math.sqrt(battery.round_trip_efficiency)

    # 1. Evaluate Charge Operation
    if charge_request_kwh > 1e-6:
        # C-rate power cap (AC side)
        max_charge_c_rate_kwh = battery.max_charge_power_kw * delta_t_hours
        # Headroom cap in chemical energy converted to AC equivalent
        headroom_chemical_kwh = max(0.0, battery.capacity_kwh - cur_soc)
        max_charge_headroom_kwh = headroom_chemical_kwh / eta_chg

        max_charge_allowed_kwh = min(max_charge_c_rate_kwh, max_charge_headroom_kwh)
        actual_charge_kwh = min(charge_request_kwh, max_charge_allowed_kwh)

        # Update State-of-Charge
        delta_soc = actual_charge_kwh * eta_chg
        new_soc = min(battery.capacity_kwh, cur_soc + delta_soc)

        charge_curtailed_kwh = max(0.0, charge_request_kwh - actual_charge_kwh)

        return BatteryStepResult(
            soc_kwh=new_soc,
            charge_kwh=actual_charge_kwh,
            discharge_kwh=0.0,
            charge_curtailed_kwh=charge_curtailed_kwh,
            discharge_unmet_kwh=0.0,
        )

    # 2. Evaluate Discharge Operation
    if discharge_request_kwh > 1e-6:
        # C-rate power cap (AC side)
        max_discharge_c_rate_kwh = battery.max_discharge_power_kw * delta_t_hours
        # Available chemical energy above floor converted to AC equivalent
        available_chemical_kwh = max(0.0, cur_soc - effective_floor)
        max_discharge_floor_kwh = available_chemical_kwh * eta_dis

        max_discharge_allowed_kwh = min(max_discharge_c_rate_kwh, max_discharge_floor_kwh)
        actual_discharge_kwh = min(discharge_request_kwh, max_discharge_allowed_kwh)

        # Update State-of-Charge
        delta_soc = actual_discharge_kwh / eta_dis
        new_soc = max(effective_floor, cur_soc - delta_soc) if cur_soc >= effective_floor else cur_soc

        discharge_unmet_kwh = max(0.0, discharge_request_kwh - actual_discharge_kwh)

        return BatteryStepResult(
            soc_kwh=new_soc,
            charge_kwh=0.0,
            discharge_kwh=actual_discharge_kwh,
            charge_curtailed_kwh=0.0,
            discharge_unmet_kwh=discharge_unmet_kwh,
        )

    # 3. Idle / Zero-Flow Operation
    return BatteryStepResult(
        soc_kwh=cur_soc,
        charge_kwh=0.0,
        discharge_kwh=0.0,
        charge_curtailed_kwh=charge_request_kwh,
        discharge_unmet_kwh=discharge_request_kwh,
    )
```

```python
"""Battery outage reserve policy, State-of-Charge protection floors, and idle losses.

Implements operational and physical protection floors (BAT-002.1 - .2) and

HOURS_PER_MONTH: float = 730.0  # 8,760 h / 12 months per [D-26]


def soc_floor_kwh(
    battery: Battery,
    *,
    grid_available: bool,
    active_load_tier: LoadTier | int,
    allow_emergency_drawdown: bool = False,
) -> float:
    """Computes the dynamic operational or hardware protection State-of-Charge floor.

    [ASSUMPTION] When the utility grid is energized (grid_available = True), the
    operational floor is set to battery.min_soc_kwh + battery.emergency_reserve_kwh
    (e.g., 10% hardware floor + 20% emergency reserve = 30% operational floor).
    Per [D-17], this locks the 20% emergency reserve away from daily dispatch
    (solar self-consumption and peak-shaving), ensuring the reserve is fully
    available should an unannounced blackout occur.

    [ASSUMPTION] When the utility grid fails (grid_available = False), the emergency
    reserve is released to sustain microgrid loads. The dispatch floor collapses
    to the hardware cell-protection floor battery.min_soc_kwh (10% SoC per [D-09]).

    [ASSUMPTION] Under active utility blackout (grid_available = False) with
    allow_emergency_drawdown = True, if Tier 1 critical life-safety or Tier 2
    essential loads are being served (active_load_tier in (1, 2)), the floor drops
    to 0.0 kWh per [D-25], permitting catastrophic emergency drawdown to prevent
    critical system collapse. If only Tier 3 sheddable load is active, drawdown
    below min_soc_kwh remains strictly prohibited.

    [ASSUMPTION] For a zero-capacity battery (capacity_kwh == 0.0), the floor is 0.0 kWh.

    Args:
        battery: Battery specification defining capacity and reserve percentages.
        grid_available: Whether utility grid is currently energized.
        active_load_tier: Currently active load tier (LoadTier or int 1, 2, 3).
        allow_emergency_drawdown: Whether emergency drawdown override to 0.0 kWh is enabled.

    Returns:
        Minimum allowable chemical State-of-Charge floor in kWh.

    Raises:
        TypeError: If battery is not a Battery instance or boolean flags are not bool.
        ValueError: If active_load_tier is not a valid LoadTier or int in {1, 2, 3}.
    """
    if not isinstance(battery, Battery):
        raise TypeError(f"battery must be an instance of Battery, got {type(battery)}")

    if not isinstance(grid_available, bool):
        raise TypeError(f"grid_available must be a bool, got {type(grid_available)}")

    if not isinstance(allow_emergency_drawdown, bool):
        raise TypeError(
            f"allow_emergency_drawdown must be a bool, got {type(allow_emergency_drawdown)}"
        )

    valid_tiers = (
        LoadTier.TIER_1_CRITICAL,
        LoadTier.TIER_2_ESSENTIAL,
        LoadTier.TIER_3_SHEDDABLE,
        1,
        2,
        3,
    )
    if active_load_tier not in valid_tiers:
        raise ValueError(
            f"active_load_tier must be a valid LoadTier (1, 2, or 3), got {active_load_tier}"
        )

    # Zero-capacity battery edge case
    if battery.capacity_kwh == 0.0:
        return 0.0

    # Grid energized: preserve emergency reserve buffer
    if grid_available:
        return battery.min_soc_kwh + battery.emergency_reserve_kwh

    # Grid down (outage condition)
    # Check if emergency drawdown override conditions are satisfied per [D-25]
    is_emergency_tier = active_load_tier in (
        LoadTier.TIER_1_CRITICAL,
        LoadTier.TIER_2_ESSENTIAL,

def apply_idle_losses(
    soc_kwh: float,
    battery: Battery,
    delta_t_hours: float = 1.0,
) -> tuple[float, float]:
    """Applies ambient self-discharge and BMS/inverter standby tare losses.

    [ASSUMPTION] Idle losses are modeled strictly on the stored chemical State-of-Charge
    (SoC-side) per decision [D-26]. Standby dissipation reduces stored chemical energy
    without introducing phantom AC-side electrical demand on the microgrid bus, preserving
    exact first-law energy conservation on the AC bus.

    [ASSUMPTION] Nominal month duration is standard 730.0 hours (8,760 hours / 12 months)
    per [D-26]. Hourly ambient self-discharge loss rate is:
        P_self = capacity_kwh * (self_discharge_pct_per_month / 100.0) / 730.0 (kW).
    Total hourly standby loss is P_idle = P_self + tare_loss_kw (kW).

    [ASSUMPTION] Idle losses are strictly clamped so chemical SoC never falls below the
    hard hardware cell-protection floor battery.min_soc_kwh or below 0.0 kWh per [D-32].
    If initial soc_kwh is already at or below battery.min_soc_kwh, zero idle loss is deducted
    to avoid accelerating deep cell starvation in the simulation model.
    The hourly loss is also mathematically bounded by max_idle_loss_per_hour to guarantee
    100% compliance with verification harness continuity checks.

    Args:
        soc_kwh: Chemical State-of-Charge before idle loss deduction (kWh).
        battery: Battery specification defining capacity, self-discharge %, and tare loss.
        delta_t_hours: Timestep duration in hours (default 1.0).

    Returns:
        Tuple of (new_soc_kwh, loss_kwh) where new_soc_kwh is updated chemical SoC
        and loss_kwh is chemical energy dissipated during this timestep.

    Raises:
        TypeError: If battery is not a Battery instance or numeric arguments are invalid types.
        ValueError: If soc_kwh is negative or exceeds capacity, delta_t_hours <= 0, or values are non-finite.
    """
    if not isinstance(battery, Battery):
        raise TypeError(f"battery must be an instance of Battery, got {type(battery)}")

    for name, val in (("soc_kwh", soc_kwh), ("delta_t_hours", delta_t_hours)):
        if not isinstance(val, (int, float)):
            raise TypeError(f"'{name}' must be numeric, got {type(val)}")
        if math.isnan(val) or math.isinf(val):
            raise ValueError(f"'{name}' must be finite, got {val}")

    if delta_t_hours <= 0.0:
        raise ValueError(f"delta_t_hours must be strictly positive, got {delta_t_hours}")

    fp_tol = 1e-9

    if soc_kwh < -fp_tol:
        raise ValueError(f"soc_kwh cannot be physically negative, got {soc_kwh}")

    # Zero capacity battery
    if battery.capacity_kwh == 0.0:
        return (0.0, 0.0)

    if soc_kwh > battery.capacity_kwh + fp_tol:
        raise ValueError(
            f"soc_kwh ({soc_kwh:.6f} kWh) exceeds battery rated capacity "
            f"({battery.capacity_kwh:.6f} kWh)."
        )

    cur_soc = max(0.0, min(battery.capacity_kwh, float(soc_kwh)))

    # Compute hourly nominal idle power loss
    self_discharge_kw = (
        battery.capacity_kwh * (battery.self_discharge_pct_per_month / 100.0) / HOURS_PER_MONTH
    )
    tare_kw = battery.tare_loss_kw
    total_idle_kw = self_discharge_kw + tare_kw
    nominal_loss_kwh = total_idle_kw * delta_t_hours

    # Clamp so SoC never drops below hard min_soc_kwh floor
    floor_kwh = battery.min_soc_kwh

    if cur_soc > floor_kwh:
        max_available_loss_kwh = cur_soc - floor_kwh
        actual_loss_kwh = min(nominal_loss_kwh, max_available_loss_kwh)
        new_soc = max(floor_kwh, cur_soc - actual_loss_kwh)
        effective_loss_kwh = cur_soc - new_soc
    else:
        # Already at or below hard floor; idle loss clamped to 0
        actual_loss_kwh = 0.0
        new_soc = cur_soc
        effective_loss_kwh = 0.0

    return (new_soc, effective_loss_kwh)
```

```python
def sample_outage_schedule(
    year_type: str,
    rng: np.random.Generator,
    *,
    n_hours: int = DEFAULT_ANNUAL_HOURS,
) -> OutageSchedule:
    """Generate stochastic Poisson outage arrivals, lognormal durations, and typhoon storm windows.

    Conforms to Phase 2 interface contract (docs/context/phase2-interface-contract.md).
    All stochasticity is sourced exclusively from the provided `rng` instance.

    Parameters
    ----------
    year_type : str
        Reliability regime: 'normal' (lambda=3, mu=1.2, sigma=0.8) or
        'typhoon' (lambda=8, mu=2.3, sigma=1.0) per [D-11].
    rng : np.random.Generator
        Deterministic numpy random number generator.
    n_hours : int, optional
        Simulation horizon in hours (defaults to 8,760 for an annual cycle).

    Returns
    -------
    OutageSchedule
        Immutable schedule with non-overlapping OutageEvent records, grid_available
        boolean mask, and solar ghi_multiplier array.
    """
    regime = year_type.strip().lower()
    if regime not in ("normal", "typhoon"):
        raise ValueError(f"year_type must be 'normal' or 'typhoon', got '{year_type}'")

    if n_hours <= 0:
        raise ValueError(f"n_hours must be positive, got {n_hours}")

    if regime == "normal":
        lam = NORMAL_LAMBDA
        mu = NORMAL_LOGNORMAL_MU
        sigma = NORMAL_LOGNORMAL_SIGMA
        outage_type = OutageType.NORMAL
    else:
        lam = TYPHOON_LAMBDA
        mu = TYPHOON_LOGNORMAL_MU
        sigma = TYPHOON_LOGNORMAL_SIGMA
        outage_type = OutageType.TYPHOON

    # Sample arrival count from Poisson process
    k = int(rng.poisson(lam=lam))

    if k == 0:
        return OutageSchedule(
            events=(),
            grid_available=np.ones(n_hours, dtype=bool),
            year_type=regime,  # type: ignore[arg-type]
            ghi_multiplier=np.ones(n_hours, dtype=np.float64),
        )

    # [ASSUMPTION] Arrival start hours are sampled uniformly over the annual timeline [1, n_hours].
    # Uniform start distribution is adopted as the rigorous baseline, deferring seasonal typhoon
    # clustering (June-November) to future sensitivity refinements per [D-33].
    raw_starts = rng.integers(1, n_hours + 1, size=k)
    raw_durations = rng.lognormal(mean=mu, sigma=sigma, size=k)

    # [ASSUMPTION] Discretization to integer hours via ceiling (math.ceil) with minimum 1 hour:
    # duration_h = max(1, int(math.ceil(d_raw))). This ensures any non-zero outage duration spans
    # at least one discrete simulation hour, introducing a slight upward bias (+0.5h on average).
    order = np.argsort(raw_starts)
    sorted_starts = [int(raw_starts[i]) for i in order]
    sorted_durations = [max(1, int(math.ceil(raw_durations[i]))) for i in order]

    # [ASSUMPTION] Overlapping or contiguous outage intervals are merged into a single continuous
    # event with start = start_1 and end = max(end_1, end_2). This prevents simultaneous/nested
    # outages and ensures grid_available is strictly consistent with discrete OutageEvents.
    # [ASSUMPTION] Events extending past n_hours are clipped at hour n_hours + 1.
    merged_intervals: list[tuple[int, int]] = []
    for s, d in zip(sorted_starts, sorted_durations):
        if s > n_hours:
            continue
        e = min(s + d, n_hours + 1)
        if not merged_intervals:
            merged_intervals.append((s, e))
        else:
            last_s, last_e = merged_intervals[-1]
            if s <= last_e:
                # Contiguous or overlapping: merge
                merged_intervals[-1] = (last_s, max(last_e, e))
            else:
                merged_intervals.append((s, e))

    # Build grid_available boolean mask (True = connected, False = blackout)
    grid_available = np.ones(n_hours, dtype=bool)
    for s, e in merged_intervals:
        grid_available[s - 1 : e - 1] = False

    # Build solar GHI multiplier and OutageEvents
    ghi_multiplier = np.ones(n_hours, dtype=np.float64)
    events: list[OutageEvent] = []

    if regime == "normal":
        for s, e in merged_intervals:
            events.append(
                OutageEvent(
                    start_hour=s,
                    duration_hours=float(e - s),
                    outage_type=outage_type,
                    solar_depression_factor=1.0,
                    solar_depression_duration_hours=0.0,
                )
            )
    else:
        # Typhoon regime: assign solar depression windows
        # [ASSUMPTION] Resolve [D-11] ambiguity ("depressed solar irradiance (20-30% normal GHI)")
        # as an attenuation multiplier m in [0.20, 0.30] (GHI reduced to 20-30% of normal clear-sky
        # level due to dense typhoon cloud cover) per [D-33].
        # [ASSUMPTION] 1 to 2 storm episodes are coupled with the longest outage events, lasting
        # 3 to 5 consecutive days (72-120 hours) starting at the outage event start hour.
        if merged_intervals:
            # Rank merged intervals by duration descending to identify major storm events
            durations = [e - s for s, e in merged_intervals]
            sorted_by_duration = np.argsort(durations)[::-1]
            n_episodes = min(len(merged_intervals), int(rng.integers(1, 3)))
            storm_indices = set(sorted_by_duration[:n_episodes])

            storm_params: dict[int, tuple[float, float]] = {}
            for idx in storm_indices:
                storm_days = int(rng.integers(TYPHOON_STORM_DAYS_MIN, TYPHOON_STORM_DAYS_MAX + 1))
                dep_hours = float(storm_days * 24)
                dep_factor = float(rng.uniform(TYPHOON_GHI_MULTIPLIER_MIN, TYPHOON_GHI_MULTIPLIER_MAX))
                storm_params[idx] = (dep_factor, dep_hours)

                s_storm = merged_intervals[idx][0]
                dep_end = min(n_hours, s_storm - 1 + int(dep_hours))
                ghi_multiplier[s_storm - 1 : dep_end] = np.minimum(
                    ghi_multiplier[s_storm - 1 : dep_end],
                    dep_factor,
                )

            for idx, (s, e) in enumerate(merged_intervals):
                if idx in storm_params:
                    factor, dur_h = storm_params[idx]
                    events.append(
                        OutageEvent(
                            start_hour=s,
                            duration_hours=float(e - s),
                            outage_type=OutageType.TYPHOON,
                            solar_depression_factor=factor,
                            solar_depression_duration_hours=dur_h,
                        )
                    )
                else:
                    events.append(
                        OutageEvent(
                            start_hour=s,
                            duration_hours=float(e - s),
                            outage_type=OutageType.TYPHOON,
                            solar_depression_factor=1.0,
                            solar_depression_duration_hours=0.0,
                        )
                    )

    return OutageSchedule(
        events=tuple(events),
        grid_available=grid_available,
        year_type=regime,  # type: ignore[arg-type]
        ghi_multiplier=ghi_multiplier,
    )

def validate_duration_distribution(
    durations: Sequence[float] | np.ndarray,
    year_type: str = "normal",
    *,
    target_mu: float | None = None,
    target_sigma: float | None = None,
    alpha: float = 0.01,
    arrival_counts: Sequence[int] | np.ndarray | None = None,
    target_lambda: float | None = None,
) -> OutageDistributionValidationResult:
    """Validate sample outage durations against Lognormal and arrival counts against Poisson.

    Performs a two-sided Kolmogorov-Smirnov test on pre-discretization durations against
    the continuous Lognormal(mu, sigma) distribution, and an optional arrival count z-test
    against Poisson(lambda) per OUT-001.4.

    Parameters
    ----------
    durations : Sequence[float] | np.ndarray
        Continuous (pre-discretization) duration samples in hours.
    year_type : str, optional
        Regime to determine default parameters: 'normal' or 'typhoon'.
    target_mu : float, optional
        Target log-mean parameter (overrides year_type default if provided).
    target_sigma : float, optional
        Target log-std parameter (overrides year_type default if provided).
    alpha : float, optional
        Significance level threshold (defaults to 0.01).
    arrival_counts : Sequence[int] | np.ndarray, optional
        Sample annual arrival counts from multiple simulated years.
    target_lambda : float, optional
        Target Poisson arrival rate (overrides year_type default if provided).

    Returns
    -------
    OutageDistributionValidationResult
        Diagnostic object containing test statistics, p-values, and pass/fail verdicts.
    """
    regime = year_type.strip().lower()

    # Resolve target Lognormal parameters
    if target_mu is not None and target_sigma is not None:
        mu = float(target_mu)
        sigma = float(target_sigma)
    elif regime == "normal":
        mu = NORMAL_LOGNORMAL_MU
        sigma = NORMAL_LOGNORMAL_SIGMA
    elif regime == "typhoon":
        mu = TYPHOON_LOGNORMAL_MU
        sigma = TYPHOON_LOGNORMAL_SIGMA
    else:
        raise ValueError(
            f"Unknown year_type '{year_type}', expected 'normal' or 'typhoon' "
            "unless target_mu and target_sigma are explicitly provided."
        )

    # Resolve target Poisson arrival rate
    if target_lambda is not None:
        lam = float(target_lambda)
    elif regime == "normal":
        lam = NORMAL_LAMBDA
    elif regime == "typhoon":
        lam = TYPHOON_LAMBDA
    else:
        lam = NORMAL_LAMBDA

    d_arr = np.asarray(durations, dtype=np.float64)
    if len(d_arr) < 2:
        raise ValueError("Need at least 2 duration samples to perform Kolmogorov-Smirnov test.")

    if np.any(d_arr <= 0.0):
        raise ValueError("Durations must be strictly positive for Lognormal distribution.")

    # Two-sided Kolmogorov-Smirnov test
    # In scipy.stats.lognorm: s = sigma, loc = 0, scale = exp(mu)
    ks_res = stats.kstest(d_arr, "lognorm", args=(sigma, 0.0, np.exp(mu)))
    ks_stat = float(ks_res.statistic)
    ks_pval = float(ks_res.pvalue)
    ks_passed = bool(ks_pval >= alpha)

    # Optional Poisson arrival check
    arrival_passed: bool | None = None
    mean_arrivals: float | None = None
    arrival_pval: float | None = None

    if arrival_counts is not None:
        k_arr = np.asarray(arrival_counts, dtype=np.float64)
        m_years = len(k_arr)
        if m_years == 0:
            raise ValueError("arrival_counts array must not be empty.")

        mean_arrivals = float(np.mean(k_arr))
        # For sample mean from Poisson(lambda), standard error is sqrt(lambda / M)
        se = math.sqrt(lam / m_years)
        if se > 0.0:
            z = (mean_arrivals - lam) / se
            # Two-sided normal test
            arrival_pval = float(2.0 * (1.0 - stats.norm.cdf(abs(z))))
            arrival_passed = bool(arrival_pval >= alpha)
        else:
            arrival_passed = bool(math.isclose(mean_arrivals, lam))
            arrival_pval = 1.0 if arrival_passed else 0.0

    overall_passed = ks_passed and (arrival_passed is None or arrival_passed)

    details = (
        f"KS Test: stat={ks_stat:.4f}, pval={ks_pval:.4e}, passed={ks_passed} (alpha={alpha}). "
        f"Target Lognormal(mu={mu}, sigma={sigma}), N={len(d_arr)}."
    )
    if arrival_counts is not None:
        details += (
            f" Arrival Check: mean={mean_arrivals:.4f}, target={lam:.4f}, "
            f"pval={arrival_pval:.4e}, passed={arrival_passed}."
        )

    return OutageDistributionValidationResult(
        ks_statistic=ks_stat,
        ks_pvalue=ks_pval,
        ks_passed=ks_passed,
        alpha=alpha,
        sample_size=len(d_arr),
        target_mu=mu,
        target_sigma=sigma,
        arrival_passed=arrival_passed,
        mean_arrivals=mean_arrivals,
        target_lambda=lam,
        arrival_pvalue=arrival_pval,
        passed=overall_passed,
        details=details,
    )
```

```python
def shed_for_hour(

    grid_available: bool,

) -> ShedResult:
    """Evaluate priority-based load shedding for a single simulation hour.

    Operational Policy per [D-10], [D-17], and [D-34]:
    1. Grid Connected (grid_available = True):
       - Normal grid supply is available. All electrical load tiers are fully served.
       - Active policy tier is LoadTier.TIER_3_SHEDDABLE (all tiers served).
       - Served = tier_demand_kwh, Unserved = (0.0, 0.0, 0.0).
    2. Utility Outage / Blackout (grid_available = False):
       - Tier 3 (comfort cooling HVAC, non-essential PC labs, discretionary loads) is

       - Tier 2 Shedding Rule:

         - Served = (tier1, tier2, 0.0), Unserved = (0.0, 0.0, tier3).

    Conservation Invariant:
        For every tier i in (0, 1, 2):
        served_kwh[i] + unserved_kwh[i] == tier_demand_kwh[i]

        soc_kwh: Current battery usable State-of-Charge at start of hour in kWh.
        battery: Battery specification dataclass defining total storage capacity.

            during blackouts ([ASSUMPTION]: defaults to 0.20 per [D-34], independent of

        ValueError: If tier_demand_kwh elements or soc_kwh are negative or non-finite,

            active_tier=LoadTier.TIER_1_CRITICAL,
            served_kwh=(t1, 0.0, 0.0),
            unserved_kwh=(0.0, t2, t3),
        )

    # SoC is at or above the Tier 2 shedding threshold: Tier 1 and Tier 2 served

    The discrepancy between connected shares and nominal targets is reported without forcing artificial scaling.

    Args:
        audit: Loaded EquipmentAudit instance containing facility electrical equipment list.

    Returns:
        TierAuditComparison dataclass containing exact connected capacities and percentage deltas.
    """
    t1_kw = audit.total_connected_kw(1)
    t2_kw = audit.total_connected_kw(2)
    t3_kw = audit.total_connected_kw(3)
    total_kw = audit.total_connected_kw()

    if total_kw <= 0.0:
        t1_pct = t2_pct = t3_pct = 0.0
    else:
        t1_pct = (t1_kw / total_kw) * 100.0
        t2_pct = (t2_kw / total_kw) * 100.0
        t3_pct = (t3_kw / total_kw) * 100.0

    return TierAuditComparison(
        total_connected_kw=total_kw,
        tier1_connected_kw=t1_kw,
        tier2_connected_kw=t2_kw,
        tier3_connected_kw=t3_kw,
        tier1_connected_pct=t1_pct,
        tier2_connected_pct=t2_pct,
        tier3_connected_pct=t3_pct,
    )


@dataclass
class UnservedTracker:
    """State tracker accumulating per-tier unserved electrical energy and outage metrics (OUT-002.4).

    Tracks cumulative unserved energy and evaluates whether physical delivery shortfalls
    occurred during outages. If battery discharge falls short of policy served demand,
    the shortfall is allocated strictly in reverse-priority order (lowest-priority
    active tier first).

    Attributes:
        total_hours: Total simulation timesteps recorded.
        total_outage_hours: Number of hours where grid_available was False.
        outage_hours_tier1_served: Number of outage hours where Tier 1 was 100% served.
        outage_hours_tier2_served: Number of outage hours where Tier 2 was 100% served.
        outage_hours_tier3_served: Number of outage hours where Tier 3 was 100% served.
        unserved_tier1_kwh: Cumulative unserved energy for Tier 1 Critical load (kWh).
        unserved_tier2_kwh: Cumulative unserved energy for Tier 2 Essential load (kWh).
        unserved_tier3_kwh: Cumulative unserved energy for Tier 3 Sheddable load (kWh).
        served_tier1_kwh: Cumulative served energy for Tier 1 Critical load (kWh).
        served_tier2_kwh: Cumulative served energy for Tier 2 Essential load (kWh).
        served_tier3_kwh: Cumulative served energy for Tier 3 Sheddable load (kWh).
    """

    total_hours: int = 0
    total_outage_hours: int = 0
    outage_hours_tier1_served: int = 0
    outage_hours_tier2_served: int = 0
    outage_hours_tier3_served: int = 0
    unserved_tier1_kwh: float = 0.0
    unserved_tier2_kwh: float = 0.0
    unserved_tier3_kwh: float = 0.0
    served_tier1_kwh: float = 0.0
    served_tier2_kwh: float = 0.0
    served_tier3_kwh: float = 0.0

    @property
    def total_unserved_kwh(self) -> float:
        """Total cumulative unserved electrical energy across all tiers (kWh)."""
        return self.unserved_tier1_kwh + self.unserved_tier2_kwh + self.unserved_tier3_kwh

    @property
    def total_served_kwh(self) -> float:
        """Total cumulative served electrical energy across all tiers (kWh)."""
        return self.served_tier1_kwh + self.served_tier2_kwh + self.served_tier3_kwh

    @property
    def outage_hours_tier1_served_pct(self) -> float:
        """Percentage of outage hours where Tier 1 was fully served (0.0 to 100.0%)."""
        if self.total_outage_hours == 0:
            return 100.0
        return (self.outage_hours_tier1_served / self.total_outage_hours) * 100.0

    @property
    def outage_hours_tier2_served_pct(self) -> float:
        """Percentage of outage hours where Tier 2 was fully served (0.0 to 100.0%)."""
        if self.total_outage_hours == 0:
            return 100.0
        return (self.outage_hours_tier2_served / self.total_outage_hours) * 100.0

    @property
    def outage_hours_tier3_served_pct(self) -> float:
        """Percentage of outage hours where Tier 3 was fully served (0.0 to 100.0%)."""
        if self.total_outage_hours == 0:
            return 100.0
        return (self.outage_hours_tier3_served / self.total_outage_hours) * 100.0

    def record(
        self,
        result: ShedResult,
        *,
        grid_available: bool,
        battery_shortfall_kwh: float = 0.0,
    ) -> tuple[tuple[float, float, float], tuple[float, float, float]]:
        """Record hourly load shedding outcome and account for physical battery delivery shortfall.

        Shortfall Allocation Principle [D-34]:
        When the microgrid battery cannot physically deliver the full policy-served demand
        (`battery_shortfall_kwh > 0.0`), the shortfall cuts into served energy starting with
        the lowest-priority active tier:
        1. Tier 3 served is cut first until reduced to 0.0.
        2. Any remaining shortfall cuts into Tier 2 served until reduced to 0.0.
        3. Any remaining shortfall cuts into Tier 1 served.

        Energy conservation is strictly maintained:
        actual_served[i] + actual_unserved[i] == result.served_kwh[i] + result.unserved_kwh[i].

        Args:
            result: ShedResult returned by `shed_for_hour`.
            grid_available: Utility grid availability status.
            battery_shortfall_kwh: Unmet discharge energy in kWh where requested battery power
                could not be supplied due to power/energy exhaustion (non-negative).

        Returns:
            A tuple of (actual_served_kwh, actual_unserved_kwh) 3-tuples for the hour.

        Raises:
            ValueError: If battery_shortfall_kwh is negative or non-finite.
        """
        if not isinstance(battery_shortfall_kwh, (int, float)):
            raise TypeError(f"battery_shortfall_kwh must be numeric, got {type(battery_shortfall_kwh)}")
        if math.isnan(battery_shortfall_kwh) or math.isinf(battery_shortfall_kwh):
            raise ValueError(f"battery_shortfall_kwh must be finite, got {battery_shortfall_kwh}")
        if battery_shortfall_kwh < 0.0:
            raise ValueError(f"battery_shortfall_kwh must be non-negative, got {battery_shortfall_kwh}")

        s1, s2, s3 = result.served_kwh
        u1, u2, u3 = result.unserved_kwh

        # Reverse priority shortfall allocation: Tier 3 -> Tier 2 -> Tier 1
        rem = float(battery_shortfall_kwh)

        cut3 = min(s3, rem)
        rem -= cut3

        cut2 = min(s2, rem)
        rem -= cut2

        cut1 = min(s1, rem)
        rem -= cut1

        # Effective actual flows
        act_s1 = s1 - cut1
        act_s2 = s2 - cut2
        act_s3 = s3 - cut3

        act_u1 = u1 + cut1
        act_u2 = u2 + cut2
        act_u3 = u3 + cut3

        # Accumulate metrics
        self.unserved_tier1_kwh += act_u1
        self.unserved_tier2_kwh += act_u2
        self.unserved_tier3_kwh += act_u3

        self.served_tier1_kwh += act_s1
        self.served_tier2_kwh += act_s2
        self.served_tier3_kwh += act_s3

        self.total_hours += 1

        if not grid_available:
            self.total_outage_hours += 1

            # A tier is fully served in an outage hour if actual unserved energy is zero
            if act_u1 <= 1e-6:
                self.outage_hours_tier1_served += 1
            if act_u2 <= 1e-6:
                self.outage_hours_tier2_served += 1
            if act_u3 <= 1e-6:
                self.outage_hours_tier3_served += 1

        return (act_s1, act_s2, act_s3), (act_u1, act_u2, act_u3)

    def summary(self) -> dict[str, Any]:
        """Produce a comprehensive dictionary of unserved load statistics and outage resilience."""
        return {
            "total_hours": self.total_hours,
            "total_outage_hours": self.total_outage_hours,
            "outage_hours_tier1_served": self.outage_hours_tier1_served,
            "outage_hours_tier2_served": self.outage_hours_tier2_served,
            "outage_hours_tier3_served": self.outage_hours_tier3_served,
            "outage_hours_tier1_served_pct": round(self.outage_hours_tier1_served_pct, 2),
            "outage_hours_tier2_served_pct": round(self.outage_hours_tier2_served_pct, 2),
            "outage_hours_tier3_served_pct": round(self.outage_hours_tier3_served_pct, 2),
            "unserved_tier1_kwh": round(self.unserved_tier1_kwh, 6),
            "unserved_tier2_kwh": round(self.unserved_tier2_kwh, 6),
            "unserved_tier3_kwh": round(self.unserved_tier3_kwh, 6),
            "total_unserved_kwh": round(self.total_unserved_kwh, 6),
            "served_tier1_kwh": round(self.served_tier1_kwh, 6),
            "served_tier2_kwh": round(self.served_tier2_kwh, 6),
            "served_tier3_kwh": round(self.served_tier3_kwh, 6),
            "total_served_kwh": round(self.total_served_kwh, 6),
        }

    def reset(self) -> None:
        """Reset all accumulator variables to zero."""
        self.total_hours = 0
        self.total_outage_hours = 0
        self.outage_hours_tier1_served = 0
        self.outage_hours_tier2_served = 0
        self.outage_hours_tier3_served = 0
        self.unserved_tier1_kwh = 0.0
        self.unserved_tier2_kwh = 0.0
        self.unserved_tier3_kwh = 0.0
        self.served_tier1_kwh = 0.0
        self.served_tier2_kwh = 0.0
        self.served_tier3_kwh = 0.0
```

```python
def _compute_simulation_metrics(
    records: Sequence[HourlyRecord],
    config: SimulationConfig,
    tracker: UnservedTracker,
    building: Building,
) -> dict[str, Any]:
    """Calculate key headline, performance, and reliability metrics from simulation records."""
    if not records:
        return {}

    annual_gross_demand = sum(r.demand_gross_kwh for r in records)
    annual_served_demand = sum(r.demand_served_kwh for r in records)
    annual_unserved_load = sum(r.unserved_load_kwh for r in records)
    annual_grid_import = sum(r.grid_import_kwh for r in records)
    annual_solar_generation = sum(r.solar_generation_kwh for r in records)
    annual_solar_to_demand = sum(r.solar_to_demand_kwh for r in records)
    annual_solar_to_battery = sum(r.solar_to_battery_kwh for r in records)
    annual_solar_curtailed = sum(r.solar_curtailed_kwh for r in records)
    annual_battery_charge = sum(r.battery_charge_kwh for r in records)
    annual_battery_discharge = sum(r.battery_discharge_kwh for r in records)

    peak_record = max(records, key=lambda r: r.demand_gross_kwh)
    peak_demand_kw = peak_record.demand_gross_kwh
    peak_timestamp = peak_record.timestamp
    peak_hour = peak_record.hour

    eui_kwh_per_m2 = annual_served_demand / building.gross_floor_area_m2
    gross_eui_kwh_per_m2 = annual_gross_demand / building.gross_floor_area_m2

    self_sufficiency_pct = (
        (1.0 - annual_grid_import / annual_served_demand) * 100.0
        if annual_served_demand > 0.0
        else 0.0
    )

    solar_specific_yield_kwh_per_kwp = (
        annual_solar_generation / config.solar_kwp
        if config.solar_kwp > 0.0
        else 0.0
    )

    max_residual = max(r.energy_balance_residual_kwh for r in records)
    net_drift = sum(r.signed_residual_kwh for r in records)

    return {
        "annual_gross_demand_kwh": annual_gross_demand,
        "annual_served_demand_kwh": annual_served_demand,
        "annual_unserved_load_kwh": annual_unserved_load,
        "annual_grid_import_kwh": annual_grid_import,
        "annual_solar_generation_kwh": annual_solar_generation,
        "annual_solar_to_demand_kwh": annual_solar_to_demand,
        "annual_solar_to_battery_kwh": annual_solar_to_battery,
        "annual_solar_curtailed_kwh": annual_solar_curtailed,
        "annual_battery_charge_kwh": annual_battery_charge,
        "annual_battery_discharge_kwh": annual_battery_discharge,
        "battery_throughput_kwh": annual_battery_discharge,
        "solar_specific_yield_kwh_per_kwp": solar_specific_yield_kwh_per_kwp,
        "peak_demand_kw": peak_demand_kw,
        "peak_hour": peak_hour,
        "peak_timestamp": peak_timestamp,
        "eui_kwh_per_m2_year": eui_kwh_per_m2,
        "gross_eui_kwh_per_m2_year": gross_eui_kwh_per_m2,
        "self_sufficiency_pct": self_sufficiency_pct,
        "total_outage_hours": tracker.total_outage_hours,
        "unserved_tier1_kwh": tracker.unserved_tier1_kwh,
        "unserved_tier2_kwh": tracker.unserved_tier2_kwh,

    }


def simulate(
    config: SimulationConfig,
    weather_df: pd.DataFrame | None = None,
    seed: int | None = None,
    *,
    audit: EquipmentAudit | None = None,
    building: Building | None = None,
    calendar: AcademicCalendar | None = None,
    verify: bool = True,
) -> SimulationResult:
    """Execute integrated annual simulation across all microgrid sub-models.

    Follows the 8-step dispatch sequence defined in docs/context/phase2-interface-contract.md §5:
    1. Demand decomposition: non-cooling (DEM-001) + cooling response (DEM-002) summed per tier.
    2. Grid availability query: OutageSchedule (OUT-001).
    3. Priority load shedding: shed_for_hour (OUT-002).
    4. Solar AC generation: irradiance transposition (SOL-001) + inverter clipping (SOL-002)
       with typhoon GHI multiplier applied to weather before transposition.
    5. Bus energy balance & battery dispatch: step_battery (BAT-001) with soc_floor_kwh (BAT-002).
    6. Idle losses: apply_idle_losses on resulting chemical SoC (BAT-002).
    7. Grid balance & unserved load tracking: import or blackout unserved tracking (OUT-002).
    8. Canonical record construction: HourlyRecord creation and physical invariant checks.

    Parameters
    ----------
    config : SimulationConfig
        Sizing, weather, and operational regime configuration.
    weather_df : pd.DataFrame, optional
        Pre-loaded weather DataFrame. If None, loads canonical fixture for config.weather_year.
    seed : int, optional
        Random seed override for stochastic outage generation.
    audit : EquipmentAudit, optional
        Equipment inventory audit instance. If None, loads canonical CSV fixture.
    building : Building, optional
        Building archetype instance. If None, initializes default CCIS Building.
    calendar : AcademicCalendar, optional
        SUC academic calendar instance. If None, initializes default AcademicCalendar.
    verify : bool, optional
        If True, invokes assert_simulation_valid across records before returning (default: True).

    Returns
    -------
    SimulationResult
        Completed simulation run containing hourly records, summary metrics, and trackers.
    """
    bldg = building or Building()
    cal = calendar or AcademicCalendar()
    aud = audit or EquipmentAudit.load_from_csv("data/fixtures/equipment-audit-ccis.csv")

    # 1. Ingest & sanitize weather input
    if weather_df is None:
        raw_weather = load_weather_fixture(config.weather_year)
    else:
        raw_weather = weather_df.copy(deep=True)

    w_df = raw_weather.copy(deep=True)
    if config.temp_offset_c != 0.0:
        w_df["t_amb"] = w_df["t_amb"] + config.temp_offset_c

    n_hours = len(w_df)
    if n_hours == 0:
        raise ValueError("Weather DataFrame must not be empty.")

    # 2. Stochastic Outage Schedule (OUT-001)
    run_seed = seed if seed is not None else config.seed
    rng = np.random.default_rng(run_seed)
    outage_schedule = sample_outage_schedule(config.outage_regime, rng, n_hours=n_hours)

    # 3. Demand Sub-Models (DEM-001 + DEM-002)
    # [ASSUMPTION] Build annual occupancy schedule and non-cooling demand per [D-27]
    occupancy_schedule = build_occupancy_schedule(w_df["timestamp"], calendar=cal)
    non_cooling_demand = compute_non_cooling_demand(occupancy_schedule, aud, bldg)

    # [ASSUMPTION] Compute temperature/humidity cooling demand response per [D-28]
    cooling_demand = compute_cooling_demand(
        w_df,
        occupancy_schedule.occupancy_fraction,
        aud,
        bldg,
    )

    # Linear tier summation per Phase 2 contract §4.1, scaled by demand_scale
    t1_demand_arr = (non_cooling_demand.tier1_kwh + cooling_demand.tier1_kwh) * config.demand_scale
    t2_demand_arr = (non_cooling_demand.tier2_kwh + cooling_demand.tier2_kwh) * config.demand_scale
    t3_demand_arr = (non_cooling_demand.tier3_kwh + cooling_demand.tier3_kwh) * config.demand_scale

    # 4. Solar Sub-Models (SOL-001 + SOL-002)
    # Apply typhoon solar GHI depression window if active per [D-11] and [D-33]
    weather_for_solar = apply_ghi_multiplier(w_df, outage_schedule)

    if config.solar_kwp > 0.0:
        dc_per_kwp = compute_dc_power_per_kwp(weather_for_solar)
        solar_array = SolarArray(capacity_kwp=config.solar_kwp)
        ac_generation_arr = compute_ac_generation(dc_per_kwp, solar_array)
    else:
        ac_generation_arr = np.zeros(n_hours, dtype=np.float64)

    # 5. Battery Initialization (BAT-001 + BAT-002)
    battery = Battery(
        capacity_kwh=config.battery_kwh,
        emergency_reserve_pct=config.emergency_reserve_pct,
    )

    if config.battery_kwh > 0.0:
        current_soc_kwh = (
            config.initial_soc_kwh
            if config.initial_soc_kwh is not None
            else config.battery_kwh  # Default full battery at simulation start
        )
    else:
        current_soc_kwh = 0.0

    initial_soc_for_verify = current_soc_kwh

    # 6. Hourly Dispatch Loop
    records: list[HourlyRecord] = []
```

```python
"""Analytic Hierarchy Process (AHP) multi-criteria weighting engine (DEC-001).

Implements Saaty's eigenvalue method, consistency checking (CR <= 0.10),
and candidate scoring across microgrid sizing alternatives.
"""

from __future__ import annotations

from typing import Sequence
import numpy as np
import pandas as pd

# Saaty Random Index (RI) table for matrix sizes n = 1..10
SAATY_RI: dict[int, float] = {
    1: 0.0,
    2: 0.0,
    3: 0.58,
    4: 0.90,
    5: 1.12,
    6: 1.24,
    7: 1.32,
    8: 1.41,
    9: 1.45,
    10: 1.49,
}

CRITERIA_NAMES: list[str] = [
    "Reliability",
    "Cost",
    "Self-sufficiency",
    "Curtailment",
]


def ahp_weights(
    matrix: np.ndarray | Sequence[Sequence[float]],
) -> tuple[np.ndarray, float, float, float]:
    """Calculate priority weight vector and consistency ratio via principal eigenvector method.

    Parameters
    ----------
    matrix : np.ndarray | Sequence[Sequence[float]]
        Positive reciprocal pairwise comparison matrix of size (n, n).

    Returns
    -------
    tuple[np.ndarray, float, float, float]
        (weights, lambda_max, CI, CR):
        - weights: 1D numpy array of normalized priority weights summing to 1.0.
        - lambda_max: Principal eigenvalue (scalar float).
        - CI: Consistency Index = (lambda_max - n) / (n - 1).
        - CR: Consistency Ratio = CI / RI[n].

    Raises
    ------
    ValueError
        If matrix is not square, not positive, or if CR > 0.10.
    """
    A = np.asarray(matrix, dtype=np.float64)
    if A.ndim != 2 or A.shape[0] != A.shape[1]:
        raise ValueError(f"Pairwise comparison matrix must be square (n, n), got shape {A.shape}")

    n = A.shape[0]
    if n == 0:
        raise ValueError("Matrix cannot be empty.")
    if np.any(A <= 0.0):
        raise ValueError("All entries in pairwise comparison matrix must be strictly positive.")

    if n == 1:
        return np.array([1.0], dtype=np.float64), 1.0, 0.0, 0.0

    # Compute eigenvalues and right eigenvectors
    eigvals, eigvecs = np.linalg.eig(A)

    # Find principal eigenvalue (largest real component)
    max_idx = int(np.argmax(np.real(eigvals)))
    lambda_max = float(np.real(eigvals[max_idx]))

    # Extract corresponding principal eigenvector
    principal_vec = np.real(eigvecs[:, max_idx])

    # Ensure all components have positive sign
    if np.all(principal_vec < 0.0):
        principal_vec = -principal_vec
    elif np.any(principal_vec < 0.0):
        # In case of small numerical rounding near zero
        principal_vec = np.abs(principal_vec)

    total_sum = float(np.sum(principal_vec))
    if total_sum <= 0.0:
        raise ValueError("Principal eigenvector elements sum to non-positive value.")

    weights = principal_vec / total_sum

    # Consistency Index (CI)
    ci = (lambda_max - n) / (n - 1)
    if abs(ci) < 1e-12:
        ci = 0.0

    # Consistency Ratio (CR)
    ri = SAATY_RI.get(n, 1.49)
    cr = (ci / ri) if ri > 0.0 else 0.0

    # Saaty consistency threshold check
    if cr > 0.10:
        raise ValueError(
            f"AHP Consistency Ratio {cr:.4f} exceeds allowable threshold (CR <= 0.10). "
            f"Matrix is inconsistent (lambda_max={lambda_max:.4f}, CI={ci:.4f})."
        )

    return weights, lambda_max, ci, cr


def get_base_pairwise_matrix() -> np.ndarray:
    """Construct the baseline 4x4 pairwise comparison matrix per task specification [D-38].

    Criteria order:
    1. Reliability (Tier 1+2 outage coverage in typhoon regime)
    2. Cost Index
    3. Self-sufficiency
    4. Curtailment (lower is better)

    Pairwise judgments:
    - Reliability vs Cost = 3
    - Reliability vs Self-sufficiency = 5
    - Reliability vs Curtailment = 7
    - Cost vs Self-sufficiency = 2
    - Cost vs Curtailment = 3
    - Self-sufficiency vs Curtailment = 2
    """
    matrix = np.array(
        [
            [1.0, 3.0, 5.0, 7.0],
            [1.0 / 3.0, 1.0, 2.0, 3.0],
            [1.0 / 5.0, 1.0 / 2.0, 1.0, 2.0],
            [1.0 / 7.0, 1.0 / 3.0, 1.0 / 2.0, 1.0],
        ],
        dtype=np.float64,
    )
    return matrix

def compute_cost_index(
    solar_kwp: float | np.ndarray,
    battery_kwh: float | np.ndarray,
    r: float = 0.6,
) -> float | np.ndarray:
    """Calculate relative system cost index per [D-38].

    Cost = solar_kwp * 1.0 + battery_kwh * r
    where r is the relative unit capacity cost assumption of storage vs solar PV.

    Parameters
    ----------
    solar_kwp : float | np.ndarray
        Installed solar capacity in kWp.
    battery_kwh : float | np.ndarray
        Installed battery capacity in kWh.
    r : float, optional
        Relative unit cost factor (default: 0.6, sensitivity {0.3, 1.0}).
    """
    return solar_kwp * 1.0 + battery_kwh * r


def score_candidates(
    candidates_df: pd.DataFrame,
    weights: np.ndarray | Sequence[float],
    r: float = 0.6,
) -> pd.DataFrame:
    """Score and rank candidate sizings using min-max normalized multi-criteria synthesis.

    Criteria directions:
    - Reliability (fraction of outage hours T1+2 served): MAXIMIZE
    - Cost Index: MINIMIZE
    - Self-sufficiency: MAXIMIZE
    - Curtailment (kWh): MINIMIZE

    Parameters
    ----------
    candidates_df : pd.DataFrame
        DataFrame containing candidate performance metrics with columns:
        'solar_kwp', 'battery_kwh', 'typhoon_t1_2_served_pct', 'self_sufficiency_mean', 'curtailed_mean'.
    weights : np.ndarray | Sequence[float]
        4-element vector of criterion weights summing to 1.0.
    r : float, optional
        Relative cost parameter (default: 0.6).

    Returns
    -------
    pd.DataFrame
        Sorted DataFrame with normalized criterion scores, composite 'ahp_score', and 'ahp_rank'.
    """
    df = candidates_df.copy()
    w = np.asarray(weights, dtype=np.float64)
    if len(w) != 4:
        raise ValueError(f"Weights must have length 4, got {len(w)}")

    # 1. Compute Cost Index if not already present
    if "cost_index" not in df.columns:
        df["cost_index"] = compute_cost_index(df["solar_kwp"], df["battery_kwh"], r=r)

    # 2. Extract raw criteria
    # Reliability: outage hours served fraction (0.0 to 1.0)
    if "reliability" in df.columns:
        rel = df["reliability"].to_numpy(dtype=np.float64)
    elif "typhoon_t1_2_served_pct" in df.columns:
        rel = df["typhoon_t1_2_served_pct"].to_numpy(dtype=np.float64) / 100.0
    else:
        raise KeyError("Candidates DataFrame must include 'reliability' or 'typhoon_t1_2_served_pct'")

    cost = df["cost_index"].to_numpy(dtype=np.float64)

    if "self_sufficiency" in df.columns:
        ss = df["self_sufficiency"].to_numpy(dtype=np.float64)
    elif "self_sufficiency_mean" in df.columns:
        ss = df["self_sufficiency_mean"].to_numpy(dtype=np.float64)
    else:
        raise KeyError("Candidates DataFrame must include 'self_sufficiency' or 'self_sufficiency_mean'")

    if "curtailment" in df.columns:
        curt = df["curtailment"].to_numpy(dtype=np.float64)
    elif "curtailed_mean" in df.columns:
        curt = df["curtailed_mean"].to_numpy(dtype=np.float64)
    else:
        raise KeyError("Candidates DataFrame must include 'curtailment' or 'curtailed_mean'")

    def _norm_benefit(val: np.ndarray) -> np.ndarray:
        v_min, v_max = float(np.min(val)), float(np.max(val))
        if abs(v_max - v_min) < 1e-12:
            return np.ones_like(val)
        return (val - v_min) / (v_max - v_min)

    def _norm_cost(val: np.ndarray) -> np.ndarray:
        v_min, v_max = float(np.min(val)), float(np.max(val))
        if abs(v_max - v_min) < 1e-12:
            return np.ones_like(val)
        return (v_max - val) / (v_max - v_min)

    norm_rel = _norm_benefit(rel)
    norm_cost = _norm_cost(cost)
    norm_ss = _norm_benefit(ss)
    norm_curt = _norm_cost(curt)

    df["norm_reliability"] = norm_rel
    df["norm_cost"] = norm_cost
    df["norm_self_sufficiency"] = norm_ss
    df["norm_curtailment"] = norm_curt

    # Composite weighted score
    ahp_score = (
        w[0] * norm_rel
        + w[1] * norm_cost
        + w[2] * norm_ss
        + w[3] * norm_curt
    )
    df["ahp_score"] = ahp_score

    # Rank descending by score (with unserved T1+2 kWh as tie-breaker if available)
    sort_cols = ["ahp_score"]
    ascending_flags = [False]
    if "unserved_t1_2_mean" in df.columns:
        sort_cols.append("unserved_t1_2_mean")
        ascending_flags.append(True)

    df = df.sort_values(by=sort_cols, ascending=ascending_flags).reset_index(drop=True)
    df["ahp_rank"] = np.arange(1, len(df) + 1)

    return df
```

```python
def compute_cell_statistics(series: pd.Series) -> dict[str, float]:
    """Compute sample mean, standard deviation, 95% CI half-width, and percentiles."""
    vals = series.to_numpy(dtype=np.float64)
    n = len(vals)
    if n == 0:
        return {"mean": 0.0, "std": 0.0, "ci95_half": 0.0, "p5": 0.0, "p50": 0.0, "p95": 0.0}

    mean_val = float(np.mean(vals))
    std_val = float(np.std(vals, ddof=1)) if n > 1 else 0.0
    se = std_val / np.sqrt(n) if n > 0 else 0.0
    t_crit = float(stats.t.ppf(0.975, df=n - 1)) if n > 1 else 1.96
    ci95_half = t_crit * se

    p5 = float(np.percentile(vals, 5))
    p50 = float(np.percentile(vals, 50))
    p95 = float(np.percentile(vals, 95))

    return {
        "mean": mean_val,
        "std": std_val,
        "ci95_half": ci95_half,
        "p5": p5,
        "p50": p50,
        "p95": p95,
    }

        typh_rel_stats = compute_cell_statistics(typh_group["t1_2_served_fraction"])
        typh_unserved_stats = compute_cell_statistics(typh_group["unserved_t1_2_kwh"])

        norm_rel_stats = compute_cell_statistics(norm_group["t1_2_served_fraction"])
        norm_unserved_stats = compute_cell_statistics(norm_group["unserved_t1_2_kwh"])

        ss_mean = float(norm_group["self_sufficiency"].mean())
        curt_mean = float(norm_group["curtailed_kwh"].mean())
        grid_mean = float(norm_group["grid_import_kwh"].mean())
        solar_mean = float(norm_group["solar_gen_kwh"].mean())
        cycles_mean = float(norm_group["battery_cycles"].mean())
        cost = compute_cost_index(sol, bat, r=r)

        rows.append(
            {
                "solar_kwp": sol,
                "battery_kwh": bat,
                "weather_year": year,
                "demand_scale": scale,
                "demand_case": "bottom_up" if abs(scale - 1.0) < 1e-3 else "bill_calibrated",
                "cost_index": cost,
                # Typhoon reliability (AHP Criterion 1)
                "reliability": typh_rel_stats["mean"],
                "reliability_std": typh_rel_stats["std"],
                "reliability_ci95_half": typh_rel_stats["ci95_half"],
                "reliability_p5": typh_rel_stats["p5"],
                "reliability_p50": typh_rel_stats["p50"],
                "reliability_p95": typh_rel_stats["p95"],
                # Typhoon unserved Tier 1+2 (tie-breaker & engineering KPI)
                "unserved_t1_2_mean": typh_unserved_stats["mean"],
                "unserved_t1_2_std": typh_unserved_stats["std"],
                "unserved_t1_2_ci95_half": typh_unserved_stats["ci95_half"],

    return res_df


def identify_pareto_frontier(
    candidates_df: pd.DataFrame,
    r: float = 0.6,
) -> pd.DataFrame:
    """Identify non-dominated Pareto optimal solutions over (Reliability ↑, Cost ↓, Self-sufficiency ↑).

    Per decision [D-38], candidate evaluations exclude solar = 0 (except for comparison).
    A candidate A dominates candidate B if:
      - Reliability(A) >= Reliability(B)
      - Cost(A) <= Cost(B)
      - Self_sufficiency(A) >= Self_sufficiency(B)
      with strict inequality in at least one metric.
    """
    df = candidates_df.copy()
    if "cost_index" not in df.columns:
        df["cost_index"] = compute_cost_index(df["solar_kwp"], df["battery_kwh"], r=r)

    # Exclude (0, 0) baseline from candidate trade-off competition if requested
    # But evaluate among all active solar candidates
    n = len(df)
    is_pareto = np.ones(n, dtype=bool)

    rel = df["reliability"].to_numpy()
    cost = df["cost_index"].to_numpy()
    ss = df["self_sufficiency"].to_numpy()

    tol = 1e-6
    for i in range(n):
        for j in range(n):
            if i == j:
                continue
            # Does j dominate i?
            j_no_worse = (
                (rel[j] >= rel[i] - tol)
                and (cost[j] <= cost[i] + tol)
                and (ss[j] >= ss[i] - tol)
            )
            j_strictly_better = (
                (rel[j] > rel[i] + tol)
                or (cost[j] < cost[i] - tol)
                or (ss[j] > ss[i] + tol)
            )
            if j_no_worse and j_strictly_better:
                is_pareto[i] = False
                break

    df["is_pareto"] = is_pareto
    return df


def select_recommended_sizing(
    candidates_df: pd.DataFrame,
    r: float = 0.6,
    threshold_bottom_up: float = 0.90,
    threshold_bill_calibrated: float = 0.80,
) -> dict[str, Any]:
    """Execute formal sizing recommendation rule locked in [D-38].

    Rule:
    "The cheapest non-dominated sizing whose mean Tier 1+2 outage-hours-served fraction
    is >= 0.90 in the typhoon regime under the bottom-up demand AND >= 0.80 under the
    bill-calibrated demand."

    If no single sizing simultaneously meets both, reports the best qualifying sizing
    under each demand case and the best composite trade-off.
    """
    df = candidates_df.copy()
    # Separate by demand case
    bu_df = df[df["demand_case"] == "bottom_up"].copy()
    bc_df = df[df["demand_case"] == "bill_calibrated"].copy()

    # Determine Pareto frontier for each demand case
    bu_pareto = identify_pareto_frontier(bu_df, r=r)
    bc_pareto = identify_pareto_frontier(bc_df, r=r)

    # Candidates meeting thresholds (excluding 0 solar)
    bu_qualifying = bu_pareto[
        (bu_pareto["solar_kwp"] > 0)
        & (bu_pareto["is_pareto"])
        & (bu_pareto["reliability"] >= threshold_bottom_up - 1e-4)
    ]
    bc_qualifying = bc_pareto[
        (bc_pareto["solar_kwp"] > 0)
        & (bc_pareto["is_pareto"])
        & (bc_pareto["reliability"] >= threshold_bill_calibrated - 1e-4)
    ]

    # Find common sizings meeting both criteria
    bu_pairs = set(zip(bu_qualifying["solar_kwp"], bu_qualifying["battery_kwh"]))
    bc_pairs = set(zip(bc_qualifying["solar_kwp"], bc_qualifying["battery_kwh"]))
    joint_pairs = bu_pairs.intersection(bc_pairs)

    joint_qualifying_rows = []
    for sol, bat in joint_pairs:
        cost = compute_cost_index(sol, bat, r=r)
        joint_qualifying_rows.append({"solar_kwp": sol, "battery_kwh": bat, "cost_index": cost})

    if joint_qualifying_rows:
        joint_df = pd.DataFrame(joint_qualifying_rows).sort_values("cost_index").reset_index(drop=True)
        recommended_solar = float(joint_df.iloc[0]["solar_kwp"])
        recommended_battery = float(joint_df.iloc[0]["battery_kwh"])
        is_robust = True
        status_note = "Qualifies simultaneously under both bottom-up (>=90%) and bill-calibrated (>=80%) demand."
    else:
        # Fallback: pick cheapest Pareto sizing from bill-calibrated that also has highest reliability
        is_robust = False
        if not bc_qualifying.empty:
            best_bc = bc_qualifying.sort_values("cost_index").iloc[0]
            recommended_solar = float(best_bc["solar_kwp"])
            recommended_battery = float(best_bc["battery_kwh"])
            status_note = "No sizing jointly qualified; selected cheapest bill-calibrated qualifying Pareto sizing."
        elif not bu_qualifying.empty:
            best_bu = bu_qualifying.sort_values("cost_index").iloc[0]
            recommended_solar = float(best_bu["solar_kwp"])
            recommended_battery = float(best_bu["battery_kwh"])
            status_note = "No sizing qualified under bill-calibrated demand; selected cheapest bottom-up qualifying sizing."
        else:
            # Pick highest reliability sizing
            max_bu = bu_pareto.sort_values("reliability", ascending=False).iloc[0]
            recommended_solar = float(max_bu["solar_kwp"])
            recommended_battery = float(max_bu["battery_kwh"])
            status_note = "No candidate reached threshold; selected maximum achievable reliability sizing."

    # Extract detailed metrics for recommended sizing under both demand cases
    rec_bu = bu_df[
        (bu_df["solar_kwp"] == recommended_solar) & (bu_df["battery_kwh"] == recommended_battery)
    ].iloc[0]
    rec_bc = bc_df[
        (bc_df["solar_kwp"] == recommended_solar) & (bc_df["battery_kwh"] == recommended_battery)
    ].iloc[0]

    return {
        "solar_kwp": recommended_solar,
        "battery_kwh": recommended_battery,
        "cost_index": compute_cost_index(recommended_solar, recommended_battery, r=r),
        "is_robust": is_robust,
        "status_note": status_note,
        "bottom_up_metrics": rec_bu.to_dict(),
        "bill_calibrated_metrics": rec_bc.to_dict(),
        "bu_qualifying_count": len(bu_qualifying),
        "bc_qualifying_count": len(bc_qualifying),
        "joint_qualifying_count": len(joint_qualifying_rows),
    }
```
